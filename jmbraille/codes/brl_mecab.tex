    \advance\brl@mecab@posnum\@ne%
  \fi%
}

\def\brl@mecab_rcheck#1#2{%
  % \def\brl@tmp{{#1}{#2}}\show\brl@tmp
  \def\brl@mecab_rcheck@cache{brl@mecab_rcheck_cache@\detokenize{#2}}%
  \ifcsname\brl@mecab_rcheck@cache\endcsname%
    \csname\brl@mecab_rcheck@cache\endcsname%
  \else%
    \brl@mecab@doubtfulfalse%
    {%
      \let\brl@mecab@prevchar\empty%
      \aftergroup\def%
      \aftergroup\brl@mecab@yomi%
      \aftergroup{%
        \brl@mecab_rcheck_#1\brl@end%
      \aftergroup}%
      \ifbrl@mecab@doubtful%
        \aftergroup\brl@mecab@doubtfultrue%
      \fi%
    }%
    \ifbrl@mecab@doubtful%
      {%
        % \def\brl@mecab_rcheck@u{う}%
        % \def\brl@mecab_rcheck@U{ウ}%
        % \xdef\brl@mecab@yomi{\brl@mecab@yomi}%
        \let\brl@text_cmd_ambb\empty%
        \let\brl@text_cmd_ambe\empty%
        \def\brl@mecab_rcheck@u{ー}% う
        \def\brl@mecab_rcheck@U{ー}% ウ
        %
        \xdef\brl@mecab@yomiothers{\brl@mecab@yomi}% こ^ちょー
      }%
    \else%
      \let\brl@mecab@yomiothers\empty%
    \fi%
    {%
      \brl@mecab_angle_amb% amb -> <>
      \xdef\brl@mecab@yomi{\brl@mecab@yomi}% こ<^ちょう>
    }%
      \expandafter\edef\csname\brl@mecab_rcheck@cache\endcsname%
    {%
      \edef\noexpand\brl@mecab@yomi{\brl@mecab@yomi}%
      \def\noexpand\brl@mecab@yomiothers{\brl@mecab@yomiothers}%
    }%
    % }%
    % \expandafter\show\csname\brl@mecab_rcheck@cache\endcsname%
    % \show\brl@mecab@yomi%
    % \show\brl@mecab@yomiothers%
  \fi%
}



\def\brl@mecab_append_prevchar{%

\def\brl@mecab_choose_yomi_#1/{%
  \ifnum\brl@mecab@readnum=\count@%

    \edef\brl@mecab@yomiothers{\brl@mecab@yomiothers/#1}%
    \advance\count@\@ne%

% determine \brl@mecab@prevdigit,\brl@mecab@prevbase
\def\brl@mecab_num_detprev#1{%
  \let\brl@mecab@prevdigit\empty%
  \let\brl@mecab@prevbase\empty%
  \ifnum#1=\z@% all zero -> 0
  \else%
    \count@\z@%
    \expandafter\brl@mecab_num_detprev_loop#1\brl@end%
    %
  \fi%
}


\def\brl@mecab_num_detprev_loop#1{%
  \def\brl@tmp{#1}%
  \ifx\brl@tmp\brl@@end%
  \else%
    \if 0#1\relax%
      \advance\count@\@ne%
      \edef\brl@mecab@prevbase{\the\count@}%
      \let\brl@mecab@prevdigit\empty%
    \else%
      \count@\z@%
      \let\brl@mecab@prevdigit\brl@tmp%
    \fi%
    \expandafter\brl@mecab_num_detprev_loop%
  \fi%
}

\def\brl@mecab_space{%
  % special case
  \edef\brl@tmp{\brl@mecab@previd _\brl@mecab@id}%
  \ifcsname brl@mecab_space@dospecial_\brl@tmp\endcsname%
    \csname brl@mecab_space@dospecial_\brl@tmp\expandafter\endcsname%
  \else%
    %
    \advance\brl@mecab@spnum\brl@mecab@prec\relax%
    \ifnum\brl@mecab@spnum>\z@%
      \def\brl@mecab_mayinsertsp{\brl@text_cmd_glue}%
    \else% no space but breakable
      \ifcsname brl@mecab_space@dospecial_\brl@mecab@id\endcsname%
        \csname brl@mecab_space@dospecial_\brl@mecab@id\endcsname%
      \else%
        \let\brl@mecab_mayinsertsp\empty% no space
      \fi%
    \fi%
  \fi%
}
