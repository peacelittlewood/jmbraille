\def\brl@document_gettextchar_exit{\brl@document_text_}%
\def\brl@document_gettextchar_exitb{\brl@document_text_bgroup}%
\def\brl@document_gettextchar_exite{\brl@document_text_egroup}%


% \def\brl@document_gettextchar#1#2#3{%
%   \def\brl@document_gettextchar_exit{#1}%
%   \def\brl@document_gettextchar_exitb{#2}%
%   \def\brl@document_gettextchar_exite{#3}%
\def\brl@document_gettextchar{%
  \let\brl@document_gettextchar@prevchar0% anything which is not a cs.
  \let\brl@document_textdo\empty%
  % \let\brl@document_gettextchar@mayinsertsp\empty% may append glue
  \brl@document_gettextchar_loop%
}
\def\brl@document_gettextchar_loop{%
  \futurelet\brl@document_gettextchar@char\brl@document_gettextchar_space%
}

\def\brl@document_gettextchar_loop_skipsp{%
  \let\brl@document_gettextchar_loop\brl@document_gettextchar_loop_orig%
  \futurelet\brl@document_gettextchar@char\brl@document_gettextchar_skipspace_loop%
}

\let\brl@document_gettextchar_loop_orig\brl@document_gettextchar_loop%


\def\brl@document_gettextchar_space{%
  \ifx\brl@document_gettextchar@char\@sptoken% check space.
    %
    % \def\brl@document_gettextchar@mayinsertsp{\brl@document_appendtoks{\brl@text_cmd_glue}}%
    %
    \brl@document_appendtoks{\brl@text_cmd_glue}%
    %
    \expandafter\brl@document_gettextchar_skipspace%
  \else%
    \expandafter\brl@document_gettextchar_group%
  \fi%
}
\let\brl@document_gettextchar_space_orig\brl@document_gettextchar_space

\expandafter\def\expandafter\brl@document_gettextchar_skipspace\brl@@space{%
  \futurelet\brl@document_gettextchar@char\brl@document_gettextchar_skipspace_loop%
}

\def\brl@document_gettextchar_skipspace_loop{%
  \ifx\brl@document_gettextchar@char\@sptoken% check space.
    \expandafter\brl@document_gettextchar_skipspace%
  \else%
    \expandafter\brl@document_gettextchar_group%
  \fi%
}

\def\brl@document_gettextchar_group{%
  %
  %\brl@document_gettextchar@mayinsertsp%
  %
  %
  \ifx\brl@document_gettextchar@char\egroup% check }.
    \def\brl@document_next{\brl@document_gettextchar_exite}%
  \else%
    \ifx\brl@document_gettextchar@char\bgroup% check {.
      \def\brl@document_next{\brl@document_gettextchar_exitb}%
    \else%
      \def\brl@document_next{\brl@document_gettextchar_char}%
    \fi%
  \fi%
  \brl@document_next%
}

\long\def\brl@document_gettextchar_char#1{%
  %\def\brl@tmp{#1}\show\brl@tmp
  \def\brl@document_next{\brl@document_gettextchar_exit#1}%
  \edef\brl@document@textchar{\string#1}%
  %
  \ifcsname brl@document_dospecial_\brl@document@textchar\endcsname% dospecial -> exit
    \edef\brl@document_textdo{\expandafter\noexpand\csname brl@document_dospecial_\brl@document@textchar\endcsname}%
  \else%
    \ifcat\relax\noexpand#1% non cs -> exit
      \ifcsname brl@code_text_\brl@document@textchar\endcsname% char -> exit
      \else%
        \ifx\brl@document@textchar\brl@document_gettextchar@prevchar% primitive
          \PackageError{jmbraille}{\brl@document@textchar\brl@@space can not be expanded. \brl@document@textchar\brl@@space may be a primitive}{}%
        \else%
          \let\brl@document_gettextchar@prevchar\brl@document@textchar% macro
          \def\brl@document_next{\expandafter\brl@document_gettextchar_loop#1}% cs & non primitive
        \fi%
      \fi%
    \fi%
  \fi%
  \brl@document_next%
}

\def\brl@document_ifequal#1#2#3#4{%
  %
  {\let\brl@text_cmd_empty\empty% {\it {\sc}} -> {\it \cmd_empty} 
    \edef\brl@ifequal@tmpa{#1}%
    \xdef\brl@ifequal@tmpa{\brl@ifequal@tmpa}% expand twice because this includes \the\toks
    \xdef\brl@ifequal@tmpb{#2}%
  }%
  %
  %
  \ifx\brl@ifequal@tmpa\brl@ifequal@tmpb% i.e., nothing is added.
    #3%
  \else%
    #4%
  \fi%
}

\def\brl@document_text_egroup_sub{%
  %
  % Caution : now {...} as the reading is obsolete.
  %
  %
  \brl@document_ifequal{\the\brl@document@toks}{\brl@document@currappendfont}{% if the contents are only fonts i.e. {\rm} etc.
    \def\brl@tmp{{\brl@text_cmd_empty}}% {}
  }{%
    \edef\brl@tmp{{\the\brl@document@toks\brl@document@currprependfont}}% 
  }%
  %
  % {\rm \bf abc ...} -> \rmopen \bfopen abc ... \bfclose \rmclose
  % or
  % {\bf ...
  % 
  % } <- This line is regarded as {}. Thus we do not need to use prependfont (finish the fonts).
  %   
  %   
  %
  \ifx\brl@document@prevtext\empty%
    % there was another paragraph in this group
    % and \brl@document@toks was already treated there.
    % Thus prevtext and toks are cleared.
    \xdef\brl@tmpcmd{\let\noexpand\brl@document@prevtext\noexpand\empty\brl@document@toks\brl@tmp}%
  \else%
    \xdef\brl@tmpcmd{\brl@document_appendtoks\brl@tmp}%
  \fi%
}

\def\brl@document_text_bgroup{%
  \afterassignment\brl@document_text_bgroup_cont%
  \let\brl@tmp% gobble \bgroup
}
  
\def\brl@document_text_bgroup_cont{%
  \bgroup%
    \edef\brl@document@prependfont{\brl@document@currprependfont\brl@document@prependfont}%
    \edef\brl@document@appendfont{\brl@document@appendfont\brl@document@currappendfont}%
    \let\brl@document@currprependfont\empty%
    \let\brl@document@currappendfont\empty%
    \edef\brl@document@prevtext{\brl@document@prevtext\the\brl@document@toks}%
    \brl@document@toks{}%
    \brl@document_text%
}

\def\brl@document_dospecial_cite[#1]#2{%
  \let\brl@document_dospecial_cite@refs\empty%
  \let\brl@document_dospecial_cite@citea\empty%
  \brl@document_dospecial_cite_#2,\brl@end,%
  {%
    \brl@document[\brl@document_dospecial_cite@refs\if@tempswa, #1\fi]\brl@document@enddoc%
    \xdef\brl@tmp{{\brl@text_cmd_noun{\the\brl@document@toks}}}%
  }%
  %
  \expandafter\brl@document_appendtoks\brl@tmp%
  \brl@document_text%
}

\def\brl@document_dospecial_cite_#1,{%
  \def\brl@tmp{#1}%
  \ifx\brl@tmp\brl@@end%
  \else%
    \edef\brl@document_dospecial_cite@citeb{\expandafter\@firstofone\brl@tmp\@empty}%
    \if@filesw\immediate\write\@auxout{\string\citation{\brl@document_dospecial_cite@citeb}}\fi%
    \@ifundefined{b@\brl@document_dospecial_cite@citeb}%
    {\def\brl@tmp{??}%
      \G@refundefinedtrue\@latex@warning{Citation `\brl@document_dospecial_cite@citeb' on page \thepage\brl@@space undefined}}%
    {\edef\brl@tmp{\csname b@\brl@document_dospecial_cite@citeb\endcsname}}%
    \edef\brl@document_dospecial_cite@refs{\brl@document_dospecial_cite@refs\brl@document_dospecial_cite@citea\brl@tmp}%
    \def\brl@document_dospecial_cite@citea{, }%
    \expandafter\brl@document_dospecial_cite_%
  \fi%
}

\def\brl@document_exec_text{%
  % finish fonts.
  %
  % if the text consists of only fonts, then add nothing.
  \brl@document_ifequal{\brl@document@prevtext\the\brl@document@toks}{\brl@document@appendfont\brl@document@currappendfont}{%
    \brl@text@toks{}%
  }{%
    \edef\brl@tmp{{\brl@document@prevtext\the\brl@document@toks\brl@document@currprependfont\brl@document@prependfont}}%
    \expandafter\brl@text\brl@tmp%
  }%
}

%% input: \brl@document@toks 
%%
%% output: \brl@text@toks 
%%         \brl@document@indent
\def\brl@document_indent{%
  %
  %
  \brl@document_exec_text%
  %
  %
  \ifbrl@document@indent%
    \global\let\brl@document@indent\brl@document@indent_default% default : 2
  \else%
    %
    \global\let\brl@document@indent\brl@document@noindent_default% default : 0
    \global\brl@document@indenttrue%
  \fi%
}

\def\brl@document_end{%
  %
  %
  \brl@document_indent%
  \brl@print{\brl@document@leftmargin}{\brl@document@indent}{\brl@text@toks}{\brl@bse@width}%
  %
  \brl@document_footnote%
  \global\brl@document@indenttrue%
  \brl@document@char% continue
}

%%
%% note : in \footnote   {   ...},
%% the first spaces are skipped while reading \footnote since it is cs.
%% the second spaces are absorbed by \brl@document_text_skipsp.
%%
\def\brl@document_dospecial_footnote{%
  \afterassignment\brl@document_dospecial_footnote_%
  \let\brl@tmp% gobble \bgroup
}

\def\brl@document_dospecial_footnote_{%
  \bgroup%
    \let\brl@document@prevtext\empty%
    \protected@edef\brl@tmp{{\brl@text_cmd_footnotemark{\@thefnmark}\brl@text_cmd_space}}%
    %
    \brl@document@toks\brl@tmp%
    \global\brl@document@indenttrue% in a footnote, indent is forced.
    % to do:
    \protected@edef\@currentlabel{%
       \csname p@footnote\endcsname\@thefnmark%
    }%
    \aftergroup\brl@document_text% \footnote{...}\brl@document_text
    \edef\brl@document@basegrouplevel{\the\currentgrouplevel}%
    \let\brl@document_text_egroup\brl@document_text_egroup_footnote%
    \let\brl@document_footnote\empty% footnote in footnote is invalidated
    \let\brl@print\brl@print_footnote% output is \brl@document@foottoks
    \brl@document_text_skipsp% skip the first spaces in \footnote{   ...}
}

\def\brl@document_getmathchar#1#2#3{%
  \def\brl@document_getmathchar_exit{#1}%
  \def\brl@document_getmathchar_exitb{#2}%
  \def\brl@document_getmathchar_exite{#3}%
  \let\brl@document_getmathchar@prevchar0% anything which is not a cs.
  \let\brl@document_mathdo\empty%
  \brl@document_getmathchar_loop%
}
\def\brl@document_getmathchar_loop{%
  \brl@futurelet\brl@document_getmathchar@char\brl@document_getmathchar_group% first skip space
}


\def\brl@document_getmathchar_group{%
  \ifx\brl@document_getmathchar@char\egroup% check }.
    \def\brl@document_next{\brl@document_getmathchar_exite}%
  \else%
    \ifx\brl@document_getmathchar@char\bgroup% check {.
      \def\brl@document_next{\brl@document_getmathchar_exitb}%
    \else%
      \def\brl@document_next{\brl@document_getmathchar_char}%
    \fi%
  \fi%
  \brl@document_next%
}


\long\def\brl@document_getmathchar_char#1{%
  \def\brl@document_next{\brl@document_getmathchar_exit#1}%
  \edef\brl@document@mathchar{\string#1}%
  \ifcsname brl@document_math_dospecial_\brl@document@mathchar\endcsname% dospecial -> exit
    \edef\brl@document_mathdo{\expandafter\noexpand\csname brl@document_math_dospecial_\brl@document@mathchar\endcsname}%
  \else%
    \ifcat\relax\noexpand#1% non cs -> exit
      \ifcsname brl@code_math_\brl@document@mathchar\endcsname% char -> exit
      \else%
        \ifx\brl@document@mathchar\brl@document_getmathchar@prevchar% primitive
          \PackageError{jmbraille}{\brl@document@mathchar\brl@@space can not be expanded. \brl@document@mathchar\brl@@space may be a primitive}{}%
        \else%
          \let\brl@document_getmathchar@prevchar\brl@document@mathchar% macro
          \def\brl@document_next{\expandafter\brl@document_getmathchar_loop#1}% cs & non primitive
        \fi%
      \fi%
    \fi%
  \fi%
  \brl@document_next%
}

\def\brl@document_footnote{%
  %
  %
  \edef\brl@tmp{\the\brl@document@mattoks}%
  \ifx\brl@tmp\empty%
  \else%
    %
    \edef\brl@tmp{%
      %
      \noexpand\brl@orig@begin{rectbox}{\brl@document_footnote@arrayelement}\noexpand\brl@document_matrix_loop%
      \the\brl@document@mattoks%
      \unexpanded{\brl@end\brl@orig@end{rectbox}}%
    }%
    \global\brl@document@mattoks{}% clear matrix elements
    \global\brl@array@matnum\z@% clear replacement of matrix elements.
    \brl@tmp%
  \fi%
  %
  %
  \edef\brl@tmp{\the\brl@document@foottoks}%
  \ifx\brl@tmp\empty%
  \else%
    \edef\brl@tmp{%
      %
      \noexpand\brl@orig@begin{rectbox}{\brl@document_footnote@footnote}\noexpand\brl@document_footnote_loop%
      \the\brl@document@foottoks%
      \unexpanded{\brl@end\brl@end\brl@end\brl@orig@end{rectbox}}%
    }%
    \global\brl@document@foottoks{}% clear footnote
    \global\brl@document@indenttrue% force indent
    \brl@tmp%
  \fi%
}


\def\brl@document_matrix_loop#1{%
  \def\brl@tmp{#1}%
  \ifx\brl@tmp\brl@@end%
  \else%
    \brl@math@toks{#1}%
    \brl@print{0}{2}{\brl@math@toks}{\brl@bse@width}%
    \expandafter\brl@document_matrix_loop%
  \fi%
}


\def\brl@document_footnote_loop#1#2#3{%
  \def\brl@tmp{#1}%
  \ifx\brl@tmp\brl@@end%
  \else%
    \brl@text@toks{#3}%
    \brl@print{#1}{#2}{\brl@text@toks}{\brl@bse@width}%
    \expandafter\brl@document_footnote_loop%
  \fi%
}


\def\brl@document_newpage{%
  \ifnum\@ne=\brl@bse@linenum% for the first page
  \else%
    \expandafter\brl@document_newpage_loop%
  \fi%
}

\def\brl@document_newpage_loop{%
  \ifnum\brl@bse@height=\brl@bse@linenum%
  \else%
    \brl@print_onescline{0}{}%
    \expandafter\brl@document_newpage_loop%
  \fi%
}

\def\brl@document_par{%
  \ifnum\brl@bse@height=\brl@bse@linenum%
  \else%
    \ifnum\@ne=\brl@bse@linenum% for the first page
    \else%
      \brl@print_onescline{0}{}%
    \fi%
  \fi%
}

\def\brl@document_nonbraille_parsepage{%
  \ifx\brl@document_nonbraille@linepage\empty%
  \else%
    %
    \expandafter\brl@document_nonbraille_parsepageb_\brl@document_nonbraille@linepage\brl@end%
    %
    %
    \expandafter\brl@document_nonbraille_parsepagee_\brl@document_nonbraille@linepage\brl@end%
    %
    \ifx\brl@document_nonbraille@pageb\brl@document_nonbraille@pagee%
      \ifx\brl@document_nonbraille@pageb\empty%
        \let\brl@document_nonbraille@pageno\empty%
      \else%
        \edef\brl@document_nonbraille@pageno{\brl@text_cmd_number{\brl@document_nonbraille@pageb}\brl@text_cmd_space}%
      \fi%
    \else%
      \ifx\brl@document_nonbraille@pageb\empty%
        \edef\brl@document_nonbraille@pageno{\brl@text_cmd_number{\brl@document_nonbraille@pagee}\brl@text_cmd_space}%
      \else%
        \edef\brl@document_nonbraille@pageno{\brl@text_cmd_number{\brl@document_nonbraille@pageb}\brl@text_cmd_space\brl@text_cmd_number{\brl@document_nonbraille@pagee}\brl@text_cmd_space}%
      \fi%
    \fi%
    \edef\brl@print@page{\brl@document_nonbraille@pageno\brl@print@page}% prepend nonbraille page
  \fi%
%
}

%% e.g.  
%% {}{208}{1}{336}{2}{451}{3}
%%
%%    〜207 page {}
%% 208〜335 page 1
%% 336〜450 page 2
%% 451〜    page 3


\def\brl@document_nonbraille_parsepageb_#1#2{%
  \def\brl@document_nonbraille@pageb{#1}%
  \def\brl@tmp{#2}%
  \ifx\brl@tmp\brl@@end%
    \let\brl@document_nonbraille_next\relax%
  \else%
    \ifnum#2>\brl@document@inputlinenob\relax%
      \def\brl@document_nonbraille_next{\brl@document_nonbraille_parsepage_end{#1}{#2}}%
    \else%
      \def\brl@document_nonbraille_next{\brl@document_nonbraille_parsepageb_}%
    \fi%
  \fi%
  \brl@document_nonbraille_next%
}

\def\brl@document_nonbraille_parsepage_end#1\brl@end{%
  \edef\brl@document_nonbraille@linepage{#1}%
}

\def\brl@document_nonbraille_parsepagee_#1#2{%
  \def\brl@document_nonbraille@pagee{#1}%
  \def\brl@tmp{#2}%
%
  \ifx\brl@tmp\brl@@end%
    \let\brl@document_nonbraille_next\relax%
  \else%
    \ifnum#2<\brl@document@inputlinenoe\relax%
      \def\brl@document_nonbraille_next{\brl@document_nonbraille_parsepagee_}%
    \else%
      \def\brl@document_nonbraille_next{\brl@document_nonbraille_parsepage_end{#1}{#2}}%
    \fi%
  \fi%
  \brl@document_nonbraille_next%
}
