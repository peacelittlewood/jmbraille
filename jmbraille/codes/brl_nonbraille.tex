\newbox\brl@nonbraille_rectbox@titlebox
\def\brl@nonbraille_rectbox#1{%
  \brl@orig@noindent%
  \bgroup%
    \setbox\brl@nonbraille_rectbox@titlebox\hbox{\space#1\space}%
    \dimen\z@\ht\brl@nonbraille_rectbox@titlebox%
    \advance\dimen\z@\dp\brl@nonbraille_rectbox@titlebox%
    \dimen\@ne\linewidth%
    \advance\dimen\@ne-20pt%
    \setbox\z@\vbox\bgroup%
      \brl@orig@vskip.5\dimen\z@%
      \brl@orig@vskip9.6pt%
      \hbox to\dimen\@ne\bgroup%
        \brl@orig@begin{minipage}{\dimen\@ne}% \linewidth-20pt
}%

\def\endbrl@nonbraille_rectbox{%
        \brl@orig@end{minipage}%
      \egroup%
      \brl@orig@vskip9.6pt%
    \egroup%
    \setbox\@ne=\vbox{%
      \offinterlineskip%
      \brl@orig@vskip.5\dimen\z@%
      \hbox to\linewidth{%
        \vbox{\hbox to 10pt{\leaders\brl@orig@hrule\@height .8pt\brl@orig@hfill}}%
        \vbox to .8pt{\vss\box\brl@nonbraille_rectbox@titlebox\vss}%
        \leaders\brl@orig@hrule\@height .8pt\brl@orig@hfill%
      }%
      \hbox to\linewidth{%
        \brl@orig@vrule\@width .8pt%
        \brl@orig@hskip -.8pt%
        \brl@orig@hfil%
        \box\z@%
        \brl@orig@hfil%
        \brl@orig@hskip -.8pt%
        \brl@orig@vrule\@width .8pt%
      }%
      \brl@orig@hrule\@height .8pt%
    }%
    \hbox{\box\@ne}%
  \egroup%
}

\def\endbrl@nonbraille_ovalbox{%
        \brl@orig@end{minipage}%
      \egroup%
    \egroup%
    \thicklines%
    \setbox\@ne\vbox{%
      \offinterlineskip%
      \brl@orig@vskip.5\dimen\z@%
      \hbox to\linewidth{%
        \vbox to -9.6pt{\hbox to 10.4pt{\brl@orig@hfill\oval(20,20)[tl]}}%
        \vbox{\hbox to 9.6pt{\leaders\brl@orig@hrule\@height .8pt\brl@orig@hfill}}%
        \vbox to .8pt{\vss\box\brl@nonbraille_rectbox@titlebox\vss}%
        \leaders\brl@orig@hrule\@height .8pt\brl@orig@hfil%
        \vbox to -9.6pt{\hbox to 10.4pt{\oval(20,20)[tr]}}%
      }%
      \brl@orig@vskip-0.8pt%
      \hbox to\linewidth{%
        \brl@orig@vrule\@width .8pt%
        \brl@orig@hskip -.8pt%
        \brl@orig@hfil%
        \box\z@%
        \brl@orig@hfil%
        \brl@orig@hskip -.8pt%
        \brl@orig@vrule\@width .8pt%
      }%
      \hbox to\linewidth{%
        \vbox to 9.6pt{\hbox to 10.4pt{\brl@orig@hfill\oval(20,20)[bl]}}%
        \leaders\brl@orig@hrule\@height\z@\@depth .8pt\brl@orig@hfil%
        \vbox to 9.6pt{\hbox to 10.4pt{\oval(20,20)[br]}}%
      }%
    }%
  \hbox{\box\@ne}%
  \egroup%
}

\def\brl@nonbraille_linepage{%
  \xdef\brl@nonbraille@inputlineno{{}{\the\inputlineno}}%
  \expandafter\brl@nonbraille_appendtoks\expandafter{\brl@nonbraille@inputlineno}%
  \let\brl@nonbraille_linepage\brl@nonbraille_linepage_next%
}

  \xdef\brl@nonbraille@inputlineno{{\thepage}{\the\inputlineno}}%
}

\def\brl@nonbraille_logread{%
  \brl@nonbraille@flushlog%
  \immediate\openin\brl@nonbraille@logfile=\jobname.log\relax%
  \brl@nonbraille_loop%
  \immediate\closein\brl@nonbraille@logfile%
}

\def\brl@nonbraille_loop{%
  \ifeof\brl@nonbraille@logfile%
    \expandafter\brl@nonbraille_logend%
  \else%
    \immediate\read\brl@nonbraille@logfile to\brl@nonbraille@logtext%
    \brl@nonbraille@overfullnum\z@%
    \expandafter\brl@nonbraille_logparse\brl@nonbraille@logtext\brl@end%
    \expandafter\brl@nonbraille_loop%
  \fi%
}

\@namedef{brl@nonbraille_logoverfull0}{O}

\long\def\brl@nonbraille_logparse#1{%
  \def\brl@tmp{#1}%
  \ifx\brl@tmp\brl@@end% end of the line
    \let\brl@nonbraille_next\relax%
  \else%
    \expandafter\ifx\csname brl@nonbraille_logoverfull\the\brl@nonbraille@overfullnum\endcsname\brl@tmp%
      \advance\brl@nonbraille@overfullnum\@ne%
      \ifnum9=\brl@nonbraille@overfullnum%
        \def\brl@nonbraille_next{\brl@nonbraille_overfull}% overfull
      \else%
        \def\brl@nonbraille_next{\brl@nonbraille_logparse}% may be overfull
      \fi%
    \else% not overfull
      \def\brl@nonbraille_next{\brl@nonbraille_skip}%
    \fi%
  \fi%
  \brl@nonbraille_next%
}
