\def\brl@array_init_width{%
  \ifnum\brl@bse@width>\count@%
    \@namedef{brl@array@maxcharnum_\the\count@}{0}%
    \@namedef{brl@array@vdat_\the\count@}{.}% default no line
    \@namedef{brl@array@vline_\the\count@}{0}% default no line
    \@namedef{brl@array@vrule_0_\the\count@}{0}%
    \advance\count@\@ne%
    \expandafter\brl@array_init_width%
  \fi%
}

\def\brl@array_init_heightdepth{%
  \ifnum\brl@bse@height_virtual>\count@%
    \@namedef{brl@array@totalheight_\the\count@}{0}%
    \@namedef{brl@array@totaldepth_\the\count@}{1}%
    \@namedef{brl@array@mul_\the\count@ _0}{0}%
    \@namedef{brl@array@hline_\the\count@}{0}% default no line
    \@namedef{brl@array@hrule_\the\count@ _0}{0}%
    \advance\count@\@ne%
    \expandafter\brl@array_init_heightdepth%
  \fi%
}

\def\brl@array_init_screen{%

    \@namedef{brl@array@content_\the\count@}{}%
    \advance\count@\@ne%
    \expandafter\brl@array_init_screen%
  \fi%
}

\def\brl@array_format_prep_rowcol#1{% format
  \def\brl@array_format@format{#1}%
  \brl@array_prep_rowcol%
}


\protected\def\brl@array@cmd_c#1#2{\expandafter\def\csname brl@array@hdat_#1\endcsname{#2}}
\protected\def\brl@array@cmd_h#1#2{\expandafter\def\csname brl@array@hline_#1\endcsname{#2}}
\protected\def\brl@array@cmd_p#1#2{\expandafter\def\csname brl@array@pele_#1\endcsname{#2}}
\protected\def\brl@array@cmd_m#1#2{\expandafter\def\csname brl@array@mul_#1\endcsname{#2}}
\protected\def\brl@array@cmd_v#1#2#3{%
  \expandafter\def\csname brl@array@vdat_#1_#2\endcsname{#3}%
  \ifnum\csname brl@array@vline_#2\endcsname<#3\relax%
    \expandafter\def\csname brl@array@vline_#2\endcsname{#3}%
  \fi%
}


\def\brl@array_prep_rowcol{%
  \bgroup% all elements are grouped. The grouping is needed because the contents of cmd_c...cmd_v above should be kept as default while \math or \text is executed in the following.
    \brl@array@toks{}% store defs
    % \brl@array@maxrownum\z@%
    \brl@array@maxcolnum\z@%
    \brl@array@rownum\z@%
    \brl@array_prep_rowcol_%
}%
\def\brl@array_prep_rowcol_end{%
    \xdef\brl@tmp{%
      \the\brl@array@toks%
      \brl@array@maxrownum\the\brl@array@rownum\relax%
      \brl@array@maxcolnum\the\brl@array@maxcolnum\relax%
    }%
  \egroup%
  \brl@tmp%
}

\def\brl@array_prep_rowcol_#1{%
  \def\brl@tmp{#1}%
  \ifx\brl@tmp\brl@@end% 
    % \brl@array@maxrownum\brl@array@rownum%
    %
    %
    % \edef\brl@array_next{%
    %   \egroup% group is closed here.
    %   \the\brl@array@toks%
    %   % \brl@array@maxrownum\the\brl@array@maxrownum\relax%
    %   \brl@array@maxrownum\the\brl@array@rownum\relax%
    %   \brl@array@maxcolnum\the\brl@array@maxcolnum\relax%
    % }%
    \def\brl@array_next{\brl@array_prep_rowcol_end}%
  \else%
    % 
    \ifx\brl@tmp\brl@array@@cline% the next consists of cline data.
      \def\brl@array_next{\brl@array_prep_rowcol_cline}%
    \else%
      \ifx\brl@tmp\brl@array@@hline%
        \count@\@ne%
        \def\brl@array_next{\brl@array_prep_rowcol_hline}%
        % 
      \else%
        % 
        \def\brl@array_next{\brl@array_prep_rowcol_}%
        % 
        \advance\brl@array@rownum\@ne%
        %
        \brl@array@colnum\z@%
        \brl@array_prep_rowcol_element#1&\brl@end&%
        \ifnum\brl@array@maxcolnum<\brl@array@colnum%
          \brl@array@maxcolnum\brl@array@colnum%
        \fi%
      \fi%
    \fi%
  \fi%
  \brl@array_next%
}%

% \brl@array@hline_(rownum) : default number of lines
% \brl@array@hdat_(rownum) : {A-B}...{X-Y}

\def\brl@array_prep_rowcol_cline#1{%
  % 
  \edef\brl@tmp{{\brl@array@cmd_c{\the\brl@array@rownum}{#1}}}%
  \expandafter\brl@array_appendtoks\brl@tmp%
  \brl@array_prep_rowcol_%
}

\def\brl@array_prep_rowcol_hline#1{%
  \def\brl@tmp{#1}%
  \ifx\brl@tmp\brl@array@@hline%
    \advance\count@\@ne%
    \expandafter\brl@array_prep_rowcol_hline%
  \else%
    \edef\brl@tmp{{\brl@array@cmd_h{\the\brl@array@rownum}{\the\count@}}}%
    \expandafter\brl@array_appendtoks\brl@tmp%
    \def\brl@tmp{\brl@array_prep_rowcol_{#1}}%
    \expandafter\brl@tmp%
  \fi%
}


% %%
% %% math mode

\def\brl@array_prep_rowcol_element_math#1&{%
  \def\brl@tmpcont{#1}%
  \ifx\brl@tmpcont\brl@@end% end line
  \else%
    \advance\brl@array@colnum\@ne%
    %
    \ifx\brl@tmpcont\empty%
    \else%
      \brl@array_prep_rowcol_element_multicolumn#1\brl@end% \brl@tmpcont may be changed.
      %
      %
      \expandafter\brl@math\expandafter{\brl@tmpcont}% math
      \edef\brl@tmp{{\brl@array@cmd_p{\the\brl@array@rownum _\the\brl@array@colnum}{\the\brl@math@toks}}}% math
      %
      %
      \expandafter\brl@array_appendtoks\brl@tmp%
    \fi%
    \expandafter\brl@array_prep_rowcol_element_math% math
  \fi%
}

%%
%% text mode

\def\brl@array_prep_rowcol_element_text#1&{%
  \def\brl@tmpcont{#1}%
  \ifx\brl@tmpcont\brl@@end% end line
  \else%
    \advance\brl@array@colnum\@ne%
    %
    \ifx\brl@tmpcont\empty% space or empty
    \else%
      \brl@array_prep_rowcol_element_multicolumn#1\brl@end% \brl@tmpcont may be changed.
      %
      %
      \expandafter\brl@text\expandafter{\brl@tmpcont}% text
      \edef\brl@tmp{{\brl@array@cmd_p{\the\brl@array@rownum _\the\brl@array@colnum}{\the\brl@text@toks}}}% text
      %
      %
      \expandafter\brl@array_appendtoks\brl@tmp%
    \fi%
    \expandafter\brl@array_prep_rowcol_element_text% text
  \fi%
}



\def\brl@array_prep_rowcol_element_multicolumn#1#2\brl@end{%
  \def\brl@tmpcmd{#1}%
  %
  %
  \edef\brl@array@mulstart{\the\brl@array@colnum}%
  \ifx\brl@tmpcmd\brl@array@@multicolumn%
    % 
    \brl@array_prep_rowcol_element_multicolumn_#2%
    % 
    % 
  \fi%
  \edef\brl@tmp{{\brl@array@cmd_m{\the\brl@array@rownum _\the\brl@array@colnum}{\brl@array@mulstart}}}%
  \expandafter\brl@array_appendtoks\brl@tmp%
}

\def\brl@array_prep_rowcol_element_multicolumn_#1#2#3{% #1: col, #2 format, #3: contents
  %
  %
  %
  \edef\brl@array@mulend{\the\numexpr\brl@array@colnum+#1\relax}%
  % 
  % 
  \brl@array_prep_rowcol_element_format#2\brl@end%
  \brl@array_prep_rowcol_element_multicolumn_loop% colnum ... colnum+#1-1
  % colnum+#1
  %
  %
  % #2 is treated here
  %
  \def\brl@tmpcont{#3}%
  %
}


\def\brl@array_prep_rowcol_element_format{%
  \count@\brl@array@colnum%
  \advance\count@\m@ne% |||c|| -> colnum-1 = 3, mulend = 2 
  \brl@array@maxcharnum\z@%
  \brl@array_prep_rowcol_element_format_loop%
}

\def\brl@array_prep_rowcol_element_format_loop#1{%
  \csname brl@array_prep_rowcol_element_format@\string#1\endcsname%
}

\@namedef{brl@array_prep_rowcol_element_format@\string\brl@end}{%
  \edef\brl@tmp{{\brl@array@cmd_v{\the\brl@array@rownum}{\the\count@}{\the\brl@array@maxcharnum}}}% overwrite
  \expandafter\brl@array_appendtoks\brl@tmp%
}

\@namedef{brl@array_prep_rowcol_element_format@|}{%
  \advance\brl@array@maxcharnum\@ne% | -> +1 
  \brl@array_prep_rowcol_element_format_loop%
}

\def\brl@array_prep_rowcol_element_format@default{% c,l,r
  \edef\brl@tmp{{\brl@array@cmd_v{\the\brl@array@rownum}{\the\count@}{\the\brl@array@maxcharnum}}}% overwrite
  \expandafter\brl@array_appendtoks\brl@tmp%
  %
  \count@\brl@array@mulend\relax%
  \brl@array@maxcharnum\z@%
  \brl@array_prep_rowcol_element_format_loop%
}
\@namedef{brl@array_prep_rowcol_element_format@c}{%
  \brl@array_prep_rowcol_element_format@default%
}
\@namedef{brl@array_prep_rowcol_element_format@r}{%
  \brl@array_prep_rowcol_element_format@default%
}
\@namedef{brl@array_prep_rowcol_element_format@l}{%
  \brl@array_prep_rowcol_element_format@default%
}
\@namedef{brl@array_prep_rowcol_element_format@p}#1{%
  \brl@array_prep_rowcol_element_format@default%
}

\def\brl@array_prep_rowcol_element_multicolumn_loop{%
  \ifnum\brl@array@colnum<\brl@array@mulend% colnum ... colnum+#1-1
%    \edef\brl@tmp{{\brl@array@cmd_p{\the\brl@array@rownum _\the\brl@array@colnum}{\noexpand\empty}%
    \edef\brl@tmp{{\brl@array@cmd_p{\the\brl@array@rownum _\the\brl@array@colnum}{}%
        \brl@array@cmd_m{\the\brl@array@rownum _\the\brl@array@colnum}{\brl@array@mulend}}}%
    \expandafter\brl@array_appendtoks\brl@tmp%
    %
    \advance\brl@array@colnum\@ne%
    \expandafter\brl@array_prep_rowcol_element_multicolumn_loop%
  \fi%
}

%% input:
%% \brl@array@maxrownum
%% \brl@array@maxcolnum
%% \brl@array@totalheight_(rownum)_(colnum) : height
%% \brl@array@totaldepth_(rownum)_(colnum) : depth
%% \brl@array@bvskip_(rownum)_(colnum) : bvskip
%% \brl@array@avskip_(rownum)_(colnum) : avskip
%%
%% output:
%% \brl@array@totalheight_(rownum) :
%% \brl@array@totaldepth_(rownum) :
%% \brl@array@totalheight : height of the array
%% \brl@array@totaldepth : depth of the array
%% \brl@array@bvskip : skip used outside the array
%% \brl@array@avskip : skip used outside the array

\brl@def{brl@array@totaldepth_1}{brl@array@totalheight_1}\brl@end{%
\def\brl@array_set_heightdepth{%
  \brl@array@rownum\@ne%
  %
  %
  \brl@array_set_heightdepth_col% the first row
  %
  % save bvskip
  \edef\brl@array@bvskip{% it is used for bvskip above the array
    \ifbrl@array@bvskip% current bvskip
      \brl@char@bvskip@true%
    \else%
      \brl@char@bvskip@false%
    \fi%
  }%
  % save avskip
  \edef\brl@array@avskip{%
    \ifbrl@array@avskip% current avskip
      \brl@char@avskip@true%
    \else%
      \brl@char@avskip@false%
    \fi%
  }%
  %
  %
  % 
  %
  %
  \edef#1{\the\brl@array@depth}%
  \edef#2{\the\brl@array@height}%
  % 
  \edef\brl@array@totaldepth{\the\numexpr\brl@array@depth+\brl@array@hdrulenum\relax}% line
  % 
  \edef\brl@array@totalheight{\the\numexpr\brl@array@height-\brl@array@hurulenum\relax}% line
  \brl@array_set_heightdepth_row%
}
}
%% 
%% 
\def\brl@array_set_heightdepth_row{%
  \advance\brl@array@rownum\@ne%
  %
  \ifnum\brl@array@rownum>\brl@array@maxrownum%
  \else%
    \brl@array_set_heightdepth_col%
    %
    % 
    % 
    \ifbrl@array@bvskip% current bvskiptrue
      \advance\brl@array@height\m@ne% vglue is counted. This does not affect in the case of the first line.
    \else%
      \brl@array@avskip% previous avskiptrue
      \ifbrl@char@avskip%
        \advance\brl@array@height\m@ne%
      \fi%
    \fi%
    %
    %
    \expandafter\edef\csname brl@array@totaldepth_\the\brl@array@rownum\endcsname{\the\brl@array@depth}%
    \expandafter\edef\csname brl@array@totalheight_\the\brl@array@rownum\endcsname{\the\brl@array@height}%
    \edef\brl@array@totaldepth{\the\numexpr\brl@array@totaldepth+\brl@array@depth-\brl@array@height\relax}%
    % 
    \edef\brl@array@avskip{\ifbrl@array@avskip\brl@char@avskip@true\else\brl@char@avskip@false\fi}% save avskip
    %
    %
    \expandafter\brl@array_set_heightdepth_row%
  \fi%
}

%%
%% \brl@array_set_heightdepth_col
%%
%% set \brl@array@height, \brl@array@depth
%% \brl@array@avskiptrue, false
%% \brl@array@bvskiptrue, false
%% in a given row.
%%

\def\brl@array_set_heightdepth_col{%
  %
  \brl@array@height\@ne% never happens  
  \brl@array@depth\m@ne% never happens
  \brl@array@colnum\@ne%
  %
  \brl@array@bvskipfalse%
  \brl@array@avskipfalse%
  % 
  \brl@array_set_heightdepth_col_%
  % 
}

% the same algorithm as that of 
% \@namedef{brl@format_oneline_\string\brl@char@block}
%
\def\brl@array_set_heightdepth_col_{%
  \edef\brl@array@rowcol{\the\brl@array@rownum _\the\brl@array@colnum}%
  \edef\brl@array@tmpheight{\csname brl@array@totalheight_\brl@array@rowcol\endcsname}%
  \edef\brl@array@tmpdepth{\csname brl@array@totaldepth_\brl@array@rowcol\endcsname}%
  %
  %
  \csname brl@array@bvskip_\brl@array@rowcol\endcsname%
  \csname brl@array@avskip_\brl@array@rowcol\endcsname%
  % 
  \ifnum\brl@array@tmpheight>\brl@array@height%
  \else%
    \brl@array@height\brl@array@tmpheight\relax%
    \ifbrl@char@bvskip%
      \brl@array@bvskiptrue% if bvskiptrue and higher than before, bvskiptrue
    \else%
      \ifnum\brl@array@tmpheight<\brl@array@height%
        \brl@array@bvskipfalse% if bvskipfalse and strictly higher than before, avskipfalse
      \fi%
    \fi%
  \fi%
  \ifnum\brl@array@tmpdepth<\brl@array@depth%
  \else%
    \brl@array@depth\brl@array@tmpdepth\relax%
    \ifbrl@char@avskip%
      \brl@array@avskiptrue% if avskiptrue and deeper than before, avskiptrue
    \else%
      \ifnum\brl@array@tmpdepth>\brl@array@depth%
        \brl@array@avskipfalse% if avskipfalse and strictly deeper than before, avskipfalse
      \fi%
    \fi%
  \fi%
  % 
  %
  %
  \ifnum\brl@array@colnum<\brl@array@maxcolnum%
    \advance\brl@array@colnum\@ne%
    \expandafter\brl@array_set_heightdepth_col_%
  \fi%
}
