%% In some engine, the behavior of scalebox is strange.
%% A simple re-implementation of scalebox.
%%
%% input : scale, hbox
%% output : \brl@@scalebox@box hbox
\newbox\brl@@scalebox@box% hbox
\def\brl@@scalebox#1#2{%
  \setbox\z@\vbox{#2}%
  \setbox\tw@\copy\z@%
  \wd\z@\z@%
  \ht\z@\z@%
  \dp\z@\z@%
  \setbox\brl@@scalebox@box\hbox{%
    \brl@@pdf_bcontent%
    \brl@@pdf_invoke{#1 0 0 #1 0 0 cm}%
    \box\z@%
    \brl@@pdf_econtent%
  }%
  \wd\brl@@scalebox@box#1\wd\tw@%
  \ht\brl@@scalebox@box\z@%
  \dp\brl@@scalebox@box#1\dimexpr\ht\tw@+\dp\tw@\relax%
}

\def\brl@def_appendtoks#1{\brl@def@toks\expandafter{\the\brl@def@toks#1}}
\newtoks\brl@def@toks%

\def\brl@def{%
  \brl@def@toks{}%
  \count@\z@%
  \brl@def_%
}
\def\brl@def_#1{%
  \def\brl@tmp{#1}%
  \ifx\brl@tmp\brl@@end%
    \expandafter\brl@def__%
  \else%
    \expandafter\brl@def_appendtoks\expandafter{\csname#1\endcsname}%
    \advance\count@\@ne%
    \expandafter\brl@def_%
  \fi%
}
\def\brl@def__{%
  \afterassignment\brl@def___\toks@%
}
\def\brl@def___{%
  \edef\brl@tmp{\unexpanded{\def\brl@tmp}\csname brl@def@var_\the\count@\endcsname{\the\toks@}\noexpand\brl@tmp\the\brl@def@toks}%
  \brl@tmp%
}

%% \brl@def_fill#1#2#3
%%
%% #1 : the number to fill
%% #2 : cs to be defined
%% #3 : sth to fill 
%%
%% e.g. \brl@def_fill{3}{\cs}{a} -> \cs = `aaa'
%%
%% note that #3 is expanded
\def\brl@def_fill#1#2#3{%
  \def\brl@def@tmp{#3}%
  {%
    \count@#1\relax%
    \aftergroup\edef%
    \aftergroup#2%
    \aftergroup{%
    \brl@def_fill_loop%
    \aftergroup}%
  }%
}
\def\brl@def_fill_loop{%
  \ifnum\count@>\z@%
    \aftergroup\brl@def@tmp%
    \advance\count@\m@ne%
    \expandafter\brl@def_fill_loop%
  \fi%
}
