\def\brl@plot{%
  %
  \brl@plot_init%
  \the\brl@pgf@invoketoks%
  % 
  % treat nodes
  %
  %
  \let\brl@bc\brl@bc_plot%
  \let\brl@ebc\brl@ebc_plot%
  % dbc is never used
  \let\brl@pgf@pointsize\empty% it is sufficient if this cs is defined.
  \let\brl@char@cmd_direct\brl@@asis%
  %
  \global\setbox\brl@plot@nodebox\hbox{\kern-4bp\the\brl@pgf@nodetoks}% -4bp means 8bp/2, where 8bp is the space between two letters.
  % 
  % 
  % treat header
  % 
  % 
  \global\setbox\brl@plot@headerbox\hbox{%
    \kern-4bp%
    % 
    % 
    \brl@pgf_hbox{\numexpr\brl@pgf@minx+50\relax}{\numexpr\brl@pgf@maxy+2\relax}{\brl@plot@page}% 2 * 24 spaces = 48, 4 * 1 = 4 - 2 (1 lineskip)
    % 
    % \@namedef{brl@bc_plot@x0}{0}
    % \@namedef{brl@bc_plot@y0}{2}
    % 
  }%
  % 
  % 
  % 
  \edef\brl@plot@boundingboxx{\the\numexpr\brl@plot@maxlinex+\brl@pgf@minx-\brl@plot@leftsidemargin\relax}%
  \edef\brl@plot@boundingboxy{\the\numexpr\brl@pgf@maxy+\brl@plot@headsep-\brl@plot@maxliney\relax}%
  % 
  \brl@plot@liney\brl@pgf@maxy\relax%
  % 
  % 
  % 
  % 
  % 
  \brl@@pdf_bcontent%
  \brl@@pdf_invoke{q 0 G 0 g 0 w}%
  \brl@@pdf_invoke{q 1 0 0 1 \the\numexpr(\brl@plot@leftsidemargin-\brl@pgf@minx)*8\relax\brl@@space%
    \the\numexpr(-\brl@pgf@maxy-\brl@plot@headsep)*8\relax\brl@@space cm}%
  % 
  \brl@plot_writepts%
  \brl@@pdf_invoke{Q 1 0 0 1 -4 0 cm}%
  \brl@plot_drawrect%
  % 
  \brl@@pdf_invoke{Q}%
  \brl@@pdf_econtent%
  % 
  \brl@plot@liney\thr@@%
  \def\brl@plot_maxcmd{\brl@plot_maxcmd_a}%
  \brl@plot_sclinedef{0}%
  \brl@plot_sclinedef{1}%
  \brl@plot_sclinedef{2}%
  \brl@plot_sclinedef{3}%
  % 
  \immediate\openout\brl@plot@file=\jobname-\brl@figure@figurenum_plot.txt\relax%
  \expandafter\brl@plot_determiney\the\brl@plot@pintoks{-1}%
  \immediate\closeout\brl@plot@file%
  % 
}


\def\brl@plot_drawrect{%
  \count@\brl@plot@maxlinex_single%
  \brl@plot_drawrect_%
}

\def\brl@plot_drawrect_{%
  \brl@@pdf_invoke{\the\numexpr(\count@-\brl@plot@maxlinex_single)*8\relax\brl@@space 4 \the\numexpr\brl@plot@maxlinex_single*8\relax\brl@@space-\the\numexpr\brl@plot@maxliney*8\relax\brl@@space re S}%
  \advance\count@\brl@plot@maxlinex_single%
  \ifnum\count@<\brl@plot@maxlinex%
    \expandafter\brl@plot_drawrect_%
  \fi%
}



\def\brl@plot_writepts{%
  \ifcsname brl@pgf@minxaty_\the\brl@plot@liney\endcsname%
    \edef\brl@plot@maxxaty{\csname brl@pgf@maxxaty_\the\brl@plot@liney\endcsname}%
    \brl@plot@linex\csname brl@pgf@minxaty_\the\brl@plot@liney\endcsname\relax%
    %
    \global\expandafter\let\csname brl@pgf@minxaty_\the\brl@plot@liney\endcsname\@undefined% clear
    % 
    \ifnum\brl@plot@boundingboxy<\brl@plot@liney%
      % 
      \edef\brl@tmp{{{\the\numexpr\brl@pgf@maxy-\brl@plot@liney+\brl@plot@headsep\relax}}}% for pindisplay
      \expandafter\brl@plot_appendpintoks\brl@tmp% for pindisplay
      %
      \brl@plot_writepts_loop%
      %
      \brl@plot_appendpintoks{{-1}}% for pindisplay
    \else%
      \brl@plot_writepts_outy_loop%
    \fi%
  \fi%
  %
  % \advance\brl@plot@liney\m@ne%
  % \ifnum\brl@pgf@miny>\brl@plot@liney%
  % \else%
  \ifnum\brl@pgf@miny<\brl@plot@liney%
    \advance\brl@plot@liney\m@ne%
    \expandafter\brl@plot_writepts%
  \fi%
}

\def\brl@plot_writepts_outy_loop{% out of range of y
  \ifcsname brl@pgf@size_\the\brl@plot@linex _\the\brl@plot@liney\endcsname%
    %
    \brl@@pdf_invoke{q 1 0 0 1 \the\numexpr\brl@plot@linex*8\relax\brl@@space\the\numexpr\brl@plot@liney*8\relax\brl@@space cm}%
    %
    \brl@plot@circler% draw circle
    \brl@@pdf_invoke{Q}%
    % 
    %
    \global\expandafter\let\csname brl@pgf@size_\the\brl@plot@linex _\the\brl@plot@liney\endcsname\@undefined% clear
    %
  \fi%
  % \advance\brl@plot@linex\@ne%
  % \ifnum\brl@plot@maxxaty<\brl@plot@linex%
  % \else%
  \ifnum\brl@plot@maxxaty>\brl@plot@linex%
    \advance\brl@plot@linex\@ne%
    \expandafter\brl@plot_writepts_outy_loop%
  \fi%
}

\def\brl@plot_writepts_loop{%
  \ifcsname brl@pgf@size_\the\brl@plot@linex _\the\brl@plot@liney\endcsname%
    %
    \brl@@pdf_invoke{q 1 0 0 1 \the\numexpr\brl@plot@linex*8\relax\brl@@space\the\numexpr\brl@plot@liney*8\relax\brl@@space cm}%
    % whether out of range or not 
    \ifnum\brl@plot@boundingboxx>\brl@plot@linex%
      \brl@plot@circle% draw circle
      \edef\brl@tmp{{{\the\numexpr\brl@plot@linex-\brl@pgf@minx+\brl@plot@leftsidemargin\relax}}}% for pindisplay
      \expandafter\brl@plot_appendpintoks\brl@tmp% for pindisplay
    \else%
      \brl@plot@circler% draw circle
    \fi%
    %
    \brl@@pdf_invoke{Q}%
    %
    \global\expandafter\let\csname brl@pgf@size_\the\brl@plot@linex _\the\brl@plot@liney\endcsname\@undefined% clear
    %
  \fi%
  % \advance\brl@plot@linex\@ne%
  % \ifnum\brl@plot@maxxaty<\brl@plot@linex%
  % \else%
  \ifnum\brl@plot@maxxaty>\brl@plot@linex%
    \advance\brl@plot@linex\@ne%
    \expandafter\brl@plot_writepts_loop%
  \fi%
}



\def\brl@plot_sclinedef#1{%
  \count@\z@%
  \def\brl@tmp{#1}%
  \brl@plot_sclinedef_%
}

\def\brl@plot_sclinedef_{%
  \ifnum\count@<\brl@plot@width\relax%
    \@namedef{brl@plot@scline\the\count@ _\brl@tmp}{00}%
    \advance\count@\@ne%
    \expandafter\brl@plot_sclinedef_%
  \fi%
}

\def\brl@plot_determiney#1{%
  \def\brl@plot@curry{#1}%
  \ifnum#1=\m@ne%
    \expandafter\brl@plot_oneline_end% the last line    
  \else%
    \expandafter\brl@plot_determiney_%
  \fi%
}

\def\brl@plot_determiney_{%
  \ifnum\brl@plot@curry>\brl@plot@liney%
    \brl@plot_oneline%
    \expandafter\brl@plot_determiney_%
  \else%
    \count@\brl@plot@curry\relax%
    \divide\count@4\relax%
    \edef\brl@plot@currmody{\the\numexpr\brl@plot@curry-4*\count@\relax}%
    \expandafter\brl@plot_determinex%
  \fi%
}

\def\brl@plot_determinex#1{%
  \ifnum#1=\m@ne%
    \expandafter\brl@plot_determiney%
  \else%
    \count@#1\relax%
    \ifodd\count@%
      \let\brl@plot_setx\brl@plot_setx_second%
    \else%
      \let\brl@plot_setx\brl@plot_setx_first%
    \fi%
    \divide\count@\tw@%
    \edef\brl@tmp{\csname brl@plot@scline\the\count@ _\brl@plot@currmody\endcsname}% 00 etc.
    \expandafter\edef\csname brl@plot@scline\the\count@ _\brl@plot@currmody\endcsname{\expandafter\brl@plot_setx\brl@tmp}%
    \expandafter\brl@plot_determinex%
  \fi%
}

\def\brl@plot_setx_first#1#2{1#2}
\def\brl@plot_setx_second#1#2{#11}


\def\brl@plot_oneline{%
  \advance\brl@plot@liney\@ne%
  \brl@plot@linex\z@%
  \count@\z@%
  % 
  \brl@plot_maxcmd%
  \brl@plot_drawpoint%
  % 
  % 
  \brl@plot_sclinedef{\brl@plot@maxmody}%
}

\def\brl@plot_oneline_end{%
  \advance\brl@plot@liney\@ne%
  \brl@plot@linex\z@%
  \count@\z@%
  % 
  \brl@plot_maxcmd%
  \brl@plot_drawpoint%
  %
  %
  \ifnum\brl@plot@maxmody=\z@%
  \else%
    \brl@plot_sclinedef{\brl@plot@maxmody}%
    \expandafter\brl@plot_oneline_end%
  \fi%
}

\def\brl@plot_oneline_start{%
  {%
    \aftergroup\immediate%
    \aftergroup\write%
    \aftergroup\brl@plot@file%
    \aftergroup{%
      \expandafter\aftergroup\brl@plot@tick%
      \brl@plot_oneline_%
    \aftergroup}%
  }%
}
