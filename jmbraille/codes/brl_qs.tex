%% \brl@qs : quick sort
%%
%% \brl@qs{{12}{sign}{34}{sign}{5}{sign}{11}{sign}{4}{sign}{99}{sign}{-51}{sign}}
%% {sign} : \@ne or \m@ne
%% {num} : integer 
%%
%% output : {cs...cs}
%% \brl@qs@res = {-51}{4}{5}{11}{12}{34}{99}
%% 
%% if winding number is zero, then the corresponding xpos is output. 

\newcount\brl@qs@pivot

\def\brl@qs#1{%
  {%
    \aftergroup\brl@qs_init%
    \brl@qs_data#1\brl@end\brl@end%
  }% Here : \brl@qs_init \cs \cs ... 
  \brl@end%
  % \brl@qs_init ... \brl@end
  % 
  % 
  \@tempcnta\z@%
  \brl@qs@res%
  % 
  \let\brl@qs_choose\empty%
  \xdef\brl@qs@res{\brl@qs@res}%
}

\protected\def\brl@qs@eorule{\let\brl@qs_choose\brl@qs_choose_eorule}
\protected\def\brl@qs@zerorule{\let\brl@qs_choose\brl@qs_choose_zerorule}

\protected\def\brl@qs_choose{}%

\def\brl@qs_choose_def#1{\edef#1{\expandafter{#1}}}%
\def\brl@qs_choose_def_set_p#1{%
  \edef#1{\expandafter{#1}\@ne}%
  \let\brl@qs_choose_def\brl@qs_choose_def_set_m%
}%
\def\brl@qs_choose_def_set_m#1{%
  \edef#1{\expandafter{#1}\m@ne}%
  \let\brl@qs_choose_def\brl@qs_choose_def_set_p%
}%


\protected\def\brl@qs_choose_zerorule#1{%
  \count@\csname brl@qs@r_#1\endcsname\relax%
  \global\expandafter\let\csname brl@qs@r_#1\endcsname\@undefined%
  % 
%  \ifnum\count@=\z@% self-cancellation.
  \ifcase\count@% self-cancellation.
    \let#1\empty%
  \else%
    \@tempcntb\@tempcnta% previous
    \advance\@tempcnta\count@%
    % 
    \multiply\@tempcntb\@tempcnta%
    % 
    % \ifnum\@tempcntb=\z@%
    \ifcase\@tempcntb%
      \brl@qs_choose_def{#1}%\edef#1{\expandafter{#1}}%
    \else%
      \let#1\empty%
    \fi%
  \fi%
}

\protected\def\brl@qs_choose_eorule#1{%
  \count@\csname brl@qs@r_#1\endcsname\relax%
  \global\expandafter\let\csname brl@qs@r_#1\endcsname\@undefined%
  % 
  \ifodd\count@%
    \brl@qs_choose_def{#1}%\edef#1{\expandafter{#1}}%
  \else%
    \let#1\empty%
  \fi%
}

%
% [2,3] U [4,5] -> {2}{1}{3}{-1}{4}{1}{5}{-1} i.e. `[' <-> 1, `]' <-> -1
% 
%
\protected\def\brl@qs_choose_set#1{%
  % 
  \@tempcntb\@tempcnta% previous
  \advance\@tempcnta\csname brl@qs@r_#1\endcsname\relax%
  \global\expandafter\let\csname brl@qs@r_#1\endcsname\@undefined%
  % 
  \ifnum\@tempcntb=\brl@pgf_fill@num% previous is the max
    \edef#1{\expandafter{#1}}%
  \else%
    \ifnum\@tempcnta=\brl@pgf_fill@num% current is the max
      \edef#1{\expandafter{#1}}%
    \else%
      \let#1\empty%
    \fi%
  \fi%
  %
}

\def\brl@qs_data#1#2{%
  \def\brl@tmp{#1}%
  \ifx\brl@tmp\brl@@end%
  \else%
    \ifcsname brl@qs@r_#1\endcsname%
      \expandafter\xdef\csname brl@qs@r_#1\endcsname{\the\numexpr\csname brl@qs@r_#1\endcsname+#2\relax}% resource : winding number
    \else%
      \protected\expandafter\gdef\csname brl@qs@d_#1\endcsname{#1}%
      \expandafter\aftergroup\csname brl@qs@d_#1\endcsname%
      \expandafter\gdef\csname brl@qs@r_#1\endcsname{#2}%
    \fi%
    \expandafter\brl@qs_data%
  \fi%
}


%%
%% output: \global\brl@qs@res
%%
\def\brl@qs_init#1\brl@end{%
  \def\brl@qs@res{#1}%
  \ifx\brl@qs@res\empty%
  \else%
    {%
      {%
        \aftergroup\brl@qs_sub%
        \brl@qs_classify#1\brl@end% determine less or more
        \xdef\brl@tmp{#1}% more than or equal to pivot
      }% Here : \brl@qs_sub{pivot}{num}{num}... less than pivot
      \brl@end%
      \xdef\brl@qs@res{\brl@qs@res\brl@qs@rres}%
    }%
  \fi%
}


\def\brl@qs_sub#1{% #1 : pivot
  \expandafter\brl@qs_init\brl@tmp\brl@end% more than or equal to pivot
  \edef\brl@qs@rres{\brl@qs_choose#1\brl@qs@res}%
  \brl@qs_init%
  % {pivot}{num}{num}... \brl@end% less than pivot
}



\def\brl@qs_classify#1{%
  \brl@qs@pivot#1\relax%
  \aftergroup#1%
  \let#1\empty% delete pivot
  \brl@qs_loop%
}


\def\brl@qs_loop#1{% #1 : cs
  \ifx#1\brl@end%
  \else%
    \ifnum#1<\brl@qs@pivot%
      \aftergroup#1%
      \let#1\empty%
    \fi%
    \expandafter\brl@qs_loop%
  \fi%
}
