\ifbrl@option@cache%

\protected\def\brl@ueb#1{% instead of \mecab, the following does the corresponding process.
  \def\brl@tmp{#1}%
  \ifx\brl@tmp\empty%
  \else%
    \let\brl@tmpmat\empty%
    {%
      {%
        \let\brl@text_cmd_empty\brl@@empty%
        \def\brl@text_cmd_glue{\ }%
        %
        % use ..._save to shorten virtualchars in detokenized strings.
        %
        \let\brl@do\brl@do_save%
        \let\brl@char@block\brl@char@block_save%
        \let\brl@text_cmd_virtualchar\brl@text_cmd_virtualchar_save%
        \let\brl@char@space@true\brl@char@space@true_save%
        %
        \xdef\brl@tmp{#1}%
      }%
      \let\brl@text@csname\brl@tmp%
      \edef\brl@text@origcs{brl@text@orig_\detokenize\expandafter{\brl@text@csname}}%
      \ifcsname\brl@text@origcs\endcsname%
        \expandafter\let\expandafter\brl@tmp\csname\brl@text@origcs\endcsname%
        \immediate\write\brl@text@file{%
          \noexpand\expandafter%
          \gdef%
          \noexpand\csname%
          brl@text@orig_\noexpand\detokenize%
          {\unexpanded\expandafter{\brl@text@csname}}%
          \noexpand\endcsname%
          {\unexpanded\expandafter{\brl@tmp}}%
        }%
        \brl@tmp%
      \else%
        %
        %
        \brl@text@toks{}%
        %
        \brl@ueb_preproc#1\brl@end%
        %
        \xdef\brl@tmp{\the\brl@text@toks}%
        % 
        \xdef\brl@tmpmat{\the\brl@text@mattoks}%
        \global\brl@text@mattoks{}%
        %
        \let\brl@do\brl@do_save%
        \let\brl@char@cmd_direct\brl@char@cmd_direct_save%
        \let\brl@char@block\brl@char@block_save%
        \let\brl@char@breakable\brl@char@breakable_save%
        \let\brl@char@space@true\brl@char@space@true_save%
        %
        \immediate\write\brl@text@file{%
          \noexpand\expandafter%
          \gdef%
          \noexpand\csname%
          brl@text@orig_\noexpand\detokenize%
          {\unexpanded\expandafter{\brl@text@csname}}%
          \noexpand\endcsname%
          {%
            \xdef\noexpand\brl@tmp%
            {%
              \noexpand\brl@text_load%
              \expandafter\brl@text_save\brl@tmp\empty%
              Z%
            }%
            \ifx\brl@tmpmat\empty%
            \else%
              \xdef\noexpand\brl@tmpmat%
              {%
                \expandafter\brl@text_save_matrix\brl@tmpmat\brl@end%
              }%
            \fi%
            \ifbrl@text_appendmath@topeq%
              \noexpand\brl@text_appendmath@topeqtrue%
            \fi%
            % glue
            % \def\noexpand\brl@text@glue{\brl@text@glue}%
          }%
        }%
      \fi%
      \expandafter\brl@document_appendmattoks\expandafter{\brl@tmpmat}%
      \ifbrl@text_appendmath@topeq%
        \global\brl@document@indentfalse% to begin with an equation, force noindent.
      \fi%
    }%
  \fi%
  \brl@text@toks\expandafter{\brl@tmp}%
}

\def\brl@ueb_preproc{%
  % 
  \brl@ueb@wordtoks{}%
  \brl@ueb@alphtoks{}%
  \brl@ueb@toks{}%
  \brl@ueb@cstoks{}%
  \def\brl@ueb_next{\brl@ueb_loop}%
  %
  \def\brl@ueb@glue{1}% default glue. This is needed because the sentence may begin with $
  \def\brl@ueb@currquote{N}% quotation condition
  \let\brl@ueb@quotes\empty%
  %
  \brl@ueb_loop_ignorespaces%
}

\def\brl@ueb_loop_ignorespaces#1{%
  %
  \ifcsname brl@ueb@ignorespaces_\string#1\endcsname%
    \csname brl@ueb@ignorespaces_\string#1\expandafter\endcsname%
  \else%
    \def\brl@tmp{\brl@ueb_loop#1}%
    \expandafter\brl@tmp%
  \fi%
}

\def\brl@ueb_loop#1{%
  %
  %
  \ifcsname brl@ueb_setglue_\string#1\endcsname%
  \else%
    \def\brl@ueb@glue{1}% default glue
  \fi%
  % 
  \ifcsname brl@ueb_dospecial_\string#1\endcsname%
    \csname brl@ueb_dospecial_\string#1\expandafter\endcsname%
  \else%
    \count@\numexpr`#1\relax%
    \ifnum11903<\count@% 0x2E7F
      \ifnum40960>\count@% 0A000
        % japanese
        \brl@ueb@jptoks{#1}%
        \def\brl@ueb_next{\brl@ueb_jp}%
        %
      \else%
        \brl@ueb_appendtoks{#1}%
        \brl@ueb_appendalphtoks{#1}%
      \fi%
    \else%
      \brl@ueb_appendtoks{#1}%
      \brl@ueb_appendalphtoks{#1}%
    \fi%
    \expandafter\brl@ueb_next%
  \fi%
}

\def\brl@ueb_jp#1{%
  \def\brl@ueb_next{\brl@ueb_jp_finish#1}%
  \ifcat\relax\noexpand#1% cs?
  \else%
    \count@\numexpr`#1\relax%
    \ifnum11903<\count@% 0x2E7F
      \ifnum40960>\count@% 0A000
        % japanese
        \brl@ueb_appendjptoks{#1}%
        \def\brl@ueb_next{\brl@ueb_jp}%
      \fi%
    \fi%
  \fi%
  \brl@ueb_next%
}

\def\brl@ueb_jp_finish{%
  %
  {%
    \expandafter\brl@text_jp\expandafter{\the\brl@ueb@jptoks}%
    \brl@char_spcount{\the\brl@text@toks}%
    % \showthe\brl@text@toks
    \ifnum\brl@char@spnum>\@ne% phrase (more than two words)
      \xdef\brl@tmpjp{{{\brl@code_ueb_nonuebstart\the\brl@text@toks\brl@code_ueb_nonuebend}}}%
    \else% word
      \edef\brl@tmp{\the\brl@text@toks}%
      \def\brl@char@breakable{\brl@char@breakable_ueb}%
      \xdef\brl@tmpjp{{{\brl@code_ueb_nonuebword\brl@tmp}}}%
    \fi%
    \xdef\brl@tmpglue{\brl@text@glue}% 1 or 2
  }%
  \edef\brl@tmp{{\brl@ueb@backslash xe063}}% liblouis recognize this as cs, and insert @
  \expandafter\brl@ueb_appendtoks\brl@tmp%
  \expandafter\brl@ueb_appendcstoks\brl@tmpjp%
  \def\brl@ueb_next{\brl@ueb_loop}%
  \let\brl@ueb@glue\brl@tmpglue%
  \brl@ueb_loop\brl@text_cmd_glue%
}

}

  \ifx\brl@ueb@char\brl@ueb@@backq%
    %
    \let\brl@ueb@char\empty%

  \fi%
  \expandafter\brl@ueb_loop\brl@ueb@char%
}

\def\brl@ueb_appendchar#1{%
  \ifnum\brl@ueb@currbreak<\z@%
  % \showthe\brl@ueb@currbreak
  % \showthe\brl@ueb@currspace
    \edef\brl@tmp{{%
        \the\brl@text@toks%
        \brl@char@breakable%
        {t}%
        {\the\brl@ueb@currbreak}%
        {\the\brl@ueb@currspace}%
        %
        \the\brl@ueb@alphtoks%
        #1}}%
    \brl@ueb@alphtoks{}%
  \else%
    \edef\brl@tmp{{%
        \the\brl@text@toks%
        \the\brl@ueb@alphtoks%
        #1}}% default: no break
  \fi%
  \brl@text@toks\brl@tmp%
  \brl@ueb@alphtoks{}%
  % \showthe\brl@text@toks
  \brl@ueb@currbreak\z@% no break \brl@char@break_never\relax%
  \brl@ueb@currspace\z@%
}

\def\brl@ueb_appendcharbreaks#1#2#3{%
  \brl@ueb_appendchar{#1}%
  \brl@ueb@currbreak#2\relax%
  \brl@ueb@currspace#3\relax%  
}

\def\brl@ueb_appendbreaks#1#2{%
  \ifnum#1<\brl@ueb@currbreak%
    \brl@ueb@currbreak#1\relax%
  \fi%
  \advance\brl@ueb@currspace#2\relax%
}



\def\brl@ueb_postproc#1{%
  \edef\brl@tmp{brl@code_ueb_\string#1}%
  \ifcsname\brl@tmp\endcsname%
    \edef\brl@tmp{{\csname\brl@tmp\endcsname}}%
    \ifcsname brl@ueb_postproc@breakablechar_\string#1\endcsname%
      \expandafter\brl@ueb_appendcharbreaks\brl@tmp{\brl@char@break_middle}{0}%
    \else%
      \expandafter\brl@ueb_appendchar\brl@tmp%
    \fi%
    \expandafter\brl@ueb_postproc%
  \else%
    \csname brl@ueb_postproc_\string#1\expandafter\endcsname%
  \fi%
}
