\protected\def\brl@print#1#2#3#4{%
  \edef\brl@tmp{\the#3}%
  \ifx\brl@tmp\empty%
  \else%
    {%
      \brl@orig@everypar{}%
      \brl@format@allowspreadtrue%
      \brl@format{#1}{#2}{#3}{#4}%
      \expandafter\brl@print_allsclines_start\the\brl@format@alltoks\brl@end%
    }%
  \fi%
  \brl@print_par% enter vertical mode.
}

  \par%
}

\newcount\brl@print@scline

\def\brl@print_allsclines_start{%
  \ifbrl@print@firstvskip% previous firstvskip : if true, then vskip is skipped.
  \else%
    \brl@format@bvskip% current bvskip
    \ifbrl@char@bvskip%
      \brl@print_allsclines_insertvskip% vskip
    \else
      \ifbrl@print@avskip% previous avskip
        \brl@print_allsclines_insertvskip% vskip
      \fi%
    \fi%
  \fi%
  \brl@print_allsclines%
}

%%
%% the same as \brl@document_par
%%
\def\brl@print_allsclines_insertvskip{%
  \ifnum\brl@bse@height=\brl@bse@linenum% in the last line, no vskip.
  \else%
    \ifnum\@ne=\brl@bse@linenum% in the first line, no vskip.
    \else%
      \brl@print_onescline{0}{}%
    \fi%
  \fi%
}


\def\brl@print_allsclines_end{%
  \brl@format@avskip% current avskip
  \ifbrl@char@avskip%
    \global\brl@print@avskiptrue%
  \else%
    \global\brl@print@avskipfalse%
  \fi%
  % 
  \ifbrl@format@firstvskip% current firstvskip
    \global\brl@print@firstvskiptrue%
  \else%
    \global\brl@print@firstvskipfalse%
  \fi%
}

\def\brl@print_allsclines#1{% the total number of the sclines
  % 
  \def\brl@tmp{#1}%
  \ifx\brl@tmp\brl@@end%
    \brl@print_allsclines_end%
  \else%
    %
    \ifnum\brl@bse@height>#1\relax% if otherwise, the line exceeds the limit of a page. In this case, the newpage is in vain.
      \ifnum\brl@bse@height<\numexpr\brl@bse@linenum+#1\relax%
        \brl@document_newpage%
      \fi%
    \fi%
    % 
    %
    \brl@print@scline#1\relax%
    \expandafter\brl@print_allsclines_%
  \fi%
}


\def\brl@print_allsclines_#1#2{% #1 : width, #2 : contents
  \def\brl@tmp{#2}%
  % 
  \ifx\brl@tmp\empty% vglue
    \brl@print_allsclines_insertvskip%
  \else%
    \brl@print_onescline{#1}{#2}%
  \fi%
  \ifnum\brl@print@scline>\@ne%
    \advance\brl@print@scline\m@ne%
    \expandafter\brl@print_allsclines_%
  \else%
    \expandafter\brl@print_allsclines%
  \fi%
}


\def\brl@print_onescline#1#2{%
  \ifnum\brl@bse@height=\brl@bse@linenum%
    \brl@print_shipout%
    %
    %
    \xdef\brl@document@inputlinenob{\brl@document@inputlineno}%
    %
    \global\brl@bse@linenum\@ne%
    \global\advance\brl@bse@pagenum\@ne%
    \global\advance\brl@bse@expagenum\@ne% right page
    %
    \global\brl@print@drawtoks{}%
    \global\brl@print@exdrawtoks{}%
    \global\brl@bse@pagetoks{}%
    \global\brl@bse@expagetoks{}% right page
    \global\brl@ebse@pagetoks{}%
    \global\brl@ebse@expagetoks{}% right page
    \global\brl@Ucode@pagetoks{}%
    %
    \global\brl@bse@spreadfalse%
    %
    %
  \else%
    \xdef\brl@document@inputlinenoe{\the\inputlineno}%
    %
  \fi%
  %
  % 
  % 
  %%
  %% braille ascii
  %%
  {% \global\advance\brl@bse@linenum\@ne%
    % is done in the following because for draw, this should be after the exec
    % while for gnb, before the exec.
    % 
    \brl@print_onescline_bse{#1}{#2}%
  }%
  %
  %%
  %% Unicode
  %%
  % 
  {%
    %
    \let\brl@bc\brl@bc_Ucode% Ucode
    \let\brl@ebc\brl@ebc_Ucode% Ucode
    \let\brl@do\brl@do_code%
    \let\brl@char@cmd_direct\brl@@asis% validated
    \let\brl@char@cmd_label\@gobbletwo% invalidated
    \let\brl@char@cmd_print\@gobble% invalidated
    % only \brl@char@cmd_remark is valid.
    %
    % \brl@print@remtoks{}%
    %
    \aftergroup\brl@Ucode_appendpagetoks%
    \aftergroup{%
      #2%
      \brl@print_remark{#1}% append remark
      \aftergroup^^J%
    \aftergroup}%
    %
    %
    % 
  }%
  %
  % 
  %% 
  %% pdf
  %%
  %
  %
  %
  \brl@print@letternum\z@%
  % 
  \ifbrl@print@amb% restore ambiguity
    \global\brl@text@ambtrue%
  \else%
    \global\brl@text@ambfalse%
  \fi%
  %
  {%
    \let\brl@char@cmd_direct\brl@@asis%
    \let\brl@char@cmd_remark\@gobble% invalidated
    \global\setbox\brl@print@onepage\vtop{\unvbox\brl@print@onepage\hbox to \textwidth{\brl@orig@vrule\@width\z@\@height\brl@braille@charheight%
        \@depth\brl@braille@depth#2\hss}}%
  }%
  %
  \ifbrl@text@amb% save ambiguity
    \global\brl@print@ambtrue%
    \global\brl@text@ambfalse%
  \else%
    \global\brl@print@ambfalse%
  \fi%
  %
}

  % the followings are assumed.
  % \let\brl@dbc_appendtmptoks\brl@dbc_appendtoks%
  % \brl@bc@toks{}%
  % \brl@bc@extoks{}%
  % \brl@ebc@toks{}% for gnb

  \let\brl@dbc\brl@dbc_output%
  % \brl@bc_ebascii is never used. Instead \brl@dbc is used.

  \let\brl@char@cmd_direct\@gobble%
  \let\brl@bc@bascii_singleq\brl@bse@singleq% ' is escaped
  \let\brl@bc@bascii_bs\brl@bse@bs% \ is escaped
  %
  \brl@print@letternum\z@%
  \edef\brl@dbc@x{\the\numexpr\brl@bse@linenum*\brl@dbc@height\relax}%
  %
  % 
  #2%
  %
  %
  % 
  % output graphics
  \edef\brl@tmp{{\the\brl@dbc@toks}}%
  \expandafter\brl@print_appenddrawtoks\brl@tmp%
  %
  \edef\brl@tmp{{\brl@bse_output@linebegin\the\brl@bc@toks\brl@bse@crlf\brl@bse_output@lineend}}%
  \expandafter\brl@bse_appendpagetoks\brl@tmp%
  % 
  \ifnum#1>\brl@bse@width\relax% if this line is spread,
    \global\brl@bse@spreadtrue% this page for nab is spread
    %
    \edef\brl@tmp{{\the\brl@dbc@extoks}}%
    \expandafter\brl@print_appendexdrawtoks\brl@tmp%    
    % 
    \edef\brl@tmp{{\brl@bse_output@linebegin\the\brl@bc@extoks\brl@bse@crlf\brl@bse_output@lineend}}%
  \else%
    \edef\brl@tmp{{\brl@bse_output@linebegin\brl@bse@crlf\brl@bse_output@lineend}}% add an empty line
    %
  \fi%
  \expandafter\brl@bse_appendexpagetoks\brl@tmp%
  %
  \global\advance\brl@bse@linenum\@ne%
}

\else%


\def\brl@print_onescline_bse#1#2{%
  \global\advance\brl@bse@linenum\@ne%
  %
  % assume the followings
  % \brl@bc@toks{}%
  % \brl@bc@extoks{}%
  % \brl@ebc@toks{}% for gnb
  % \brl@ebc@extoks{}% for gnb
  % 
  %
  \let\brl@bc\brl@bc_bascii%
  \let\brl@ebc\brl@ebc_ebascii% extended bascii
  \let\brl@do\brl@do_code%
  \let\brl@char@cmd_direct\@gobble%
  \def\brl@bc@bascii_singleq{'}% '
  \let\brl@bc@bascii_bs\brl@@bs% \
  %
  \brl@print@letternum\z@%
  %
  #2%
  %
  \expandafter\brl@bse_appendpagetoks\expandafter{\the\brl@bc@toks^^J}%
  \edef\brl@tmp{\the\brl@ebc@toks}%
  \ifx\brl@tmp\empty%
  \else% since it is unclear whether this page is spread or not, tmppage is set.
    \edef\brl@tmp{{{\noexpand\brl@ebse@tmppage}{\the\brl@bse@linenum}{\brl@tmp}}}% for gnb
    \expandafter\brl@ebse_appendpagetoks\brl@tmp% for gnb
  \fi%

    \global\brl@bse@spreadtrue% this page for nab is spread
    %
    \edef\brl@tmp{\the\brl@ebc@extoks}%
    \ifx\brl@tmp\empty%
    \else%
      \edef\brl@tmp{{{\the\brl@bse@pagenum}{\the\brl@bse@linenum}{\brl@tmp}}}% for gnb
      \expandafter\brl@ebse_appendexpagetoks\brl@tmp% for gnb
    \fi%
    % 
    \expandafter\brl@bse_appendexpagetoks\expandafter{\the\brl@bc@extoks^^J}%
  \else%
    \brl@bse_appendexpagetoks{^^J}% add an empty line
  \fi%
}

\def\brl@print_remark#1{% #1 : width
  \edef\brl@tmp{{\the\brl@print@remtoks}}%
  \ifx\brl@tmp\brl@@empty%
  \else%
    \expandafter\brl@text_direct\brl@tmp%
    \count@#1\relax%
    \advance\count@\m@ne%
    \divide\count@\brl@bse@width\relax%
    \brl@format_maketspace{\the\numexpr\count@*\brl@bse@width+\brl@bse@width-#1\relax}% number to fill.
    % 
    \brl@format@fillspace% bc
    \the\brl@text@toks% bc
  \fi%
}

  \def\brl@tmp{#1}%
  \ifx\brl@tmp\brl@@end%
    %
    % \showthe\brl@plot@linex
    % \showthe\brl@plot@liney

    \multiply\brl@plot@liney\brl@plot@brailleheight\relax% height of braille

    %
    % 
    % \@namedef{brl@bc_plot@x0}{0}
    % \@namedef{brl@bc_plot@y0}{14}

    \@tempdima\brl@plot@linex\@tempdimb%

    %

  \fi%

\def\brl@print_shipout{%
  %
  % nonbraille page number 11 chars.
  %
  \edef\brl@print@page{（\brl@text_cmd_number{\brl@document@inputlinenob}\brl@text_cmd_space\brl@text_cmd_number{\brl@document@inputlinenoe}）}%
  %
  \brl@document_nonbraille_parsepage% prepend nonbraille_page if needed.
  %
  \expandafter\brl@text_direct\expandafter{\brl@print@page}%
  \brl@char_count{\the\brl@text@toks}%
  \edef\brl@tmp{{\the\numexpr26-\brl@char@charnum\relax}}%
  \expandafter\brl@format_maketspace\brl@tmp% 0,...,26
  \expandafter\brl@text_appendtoks\expandafter{\brl@format@fillspace}%
  \edef\brl@print_shipout@nonbraillepage{\the\brl@text@toks}%
  %
  %
  \brl@mecab_corpus_off%
  \edef\brl@tmp{{\brl@document@rhead}}%
  \expandafter\brl@text\brl@tmp%
  \brl@mecab_corpus_on%
  %
  %%
  %% braille ascii
  %%
  {%
    \brl@print_shipout_bse%
  }%
  %
  %%
  %% Unicode
  %%
  %
  {%
    %
    \let\brl@bc\brl@bc_Ucode%
    \let\brl@ebc\brl@ebc_Ucode%
    \let\brl@do\brl@do_code%
    \let\brl@char@cmd_direct\@gobble%
    %
    %
    \aftergroup\toks@%
    \aftergroup{%
      \brl@print_shipout@nonbraillepage%
      \the\brl@text@toks%
    \aftergroup}%
    %
  }%
  \edef\brl@tmp{{\the\toks@^^J\the\brl@Ucode@pagetoks}}%
  \expandafter\brl@Ucode_appendtoks\brl@tmp%
  % 
  %% 
  %% pdf
  %%
  %
  %
  %
  {%
    % bc and ebc are default
    %
    \let\brl@bc_braille_spread\empty% ignores brl@bse@width for bc and ebc
    \let\brl@char@cmd_direct\brl@@asis%
    \setbox\brl@print@onepage\vbox{%
      \hbox{%
        \brl@orig@vrule\@width\z@%
        \@height\brl@braille@charheight%
        \@depth\brl@braille@depth%
        %
        \brl@print_shipout@nonbraillepage\the\brl@text@toks%
      }\unvbox\brl@print@onepage}%
    \brl@print_newpage% shipout
  }%
  %
}
