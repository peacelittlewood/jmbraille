\def\brl@pgf_matrix_y{%
  \@tempdima\csname pgf@matrix@miny\the\@tempcnta\endcsname%

  \ifnum\@tempcnta>\pgfmatrixcurrentrow%
  \else%
    \expandafter\brl@pgf_matrix_y%

}

}%

\def\brl@patch_dmo*#1#2{%
  \AtBeginDocument{\brl@patch_dmo_do{#1}{#2}}%
}

\def\brl@patch@@mathcmd{\brl@math_cmd}
\def\brl@patch@@letterrK{letterrK}
\def\brl@patch@@letterrL{letterrL}


\def\brl@patch_dmo_ajsub_first#1{%
  \def\brl@tmp{#1}%
  \ifx\brl@patch@@mathcmd\brl@tmp%
    \expandafter\brl@patch_dmo_ajsub_first_%
  \else%
    \expandafter\brl@patch_dmo_ajsub_first__%
  \fi%
}

\def\brl@patch_dmo_ajsub_first_#1#2\brl@end{%
  \def\brl@tmp{#1}%
  \ifx\brl@tmp\brl@patch@@letterrK% aj
    \xdef\brl@patch@role{funcaj}%
  \else%
    \ifx\brl@tmp\brl@patch@@letterrL% kz
      \xdef\brl@patch@role{funckz}%
    \else
      \xdef\brl@patch@role{op}%
    \fi%
  \fi%
}

\def\brl@patch_dmo_ajsub_first__#1\brl@end{%
  \xdef\brl@patch@role{op}%
}



\def\brl@patch_dmo_do#1#2{%
  %
  \brl@def_cmdonly{#1}%
  %
  \brl@char_getnewkey%
  %
  % 
  {%
    {%
      \let\brl@document_math_checkalldigit\@gobble%
      \brl@document@mathtoks{}%
      \brl@document_math\rm#2\brl@document_math@end%
      %\showthe\brl@document@mathtoks
      \expandafter\brl@patch_dmo_ajsub_first\the\brl@document@mathtoks\brl@end%
      \expandafter\brl@math_parse\expandafter{\the\brl@document@mathtoks}%
 %     \xdef\brl@tmp{\the\brl@math@toks}%
    }%
    %
    % 
    \brl@math@toks\expandafter{\brl@math@subtoks}%
    %
    % 
    \brl@char@charnum\z@%
    \let\brl@do\brl@do_count%
    \the\brl@math@toks%
    %
    \edef\brl@tmp{\ifnum\brl@char@charnum>\z@\advance\brl@char@charnum\the\brl@char@charnum\relax\fi}%
    \global\expandafter\let\csname brl@char@count_\brl@char@key\endcsname\brl@tmp%
    % 
    \brl@char@spacefalse%
    \let\brl@do\brl@do_space%
    \the\brl@math@toks%
    %
    \edef\brl@tmp{\ifbrl@char@space\noexpand\brl@char@spacetrue\fi}%
    \global\expandafter\let\csname brl@char@space_\brl@char@key\endcsname\brl@tmp%
    %
    \brl@char@toks{}%
    \let\brl@do\brl@do_code%
    \let\brl@bc\brl@bc_patch%
    \let\brl@ebc\brl@ebc_patch%
    \the\brl@math@toks%
    % 
    \edef\brl@tmp{\the\brl@char@toks}%
    \global\expandafter\let\csname brl@char@code_\brl@char@key\endcsname\brl@tmp%
    \global\expandafter\let\csname brl@char@draw_\brl@char@key\endcsname\brl@tmp%
    %
    %
    \edef\brl@tmp{%
      {\brl@char_textstack{ユーザー定義}{\let\noexpand\brl@do\noexpand\brl@do_mean\the\brl@math@toks}}%
    }%
    \global\expandafter\let\csname brl@char@mean_\brl@char@key\endcsname\brl@tmp%
    % \show\brl@tmp
    %
  }%
  %
  %
  \protected\expandafter\edef\csname brl@code_math_\string#1\endcsname{\noexpand\brl@do{\brl@char@key}}%
  \expandafter\edef\csname brl@char@load_\brl@char@key\endcsname{%
    \csname brl@code_math_\string#1\endcsname}%
  %% mean
  % 
  %
  \brl@document_math_cmddef{\brl@patch@role}{#1}{\string#1}%
  %
  % 
}
