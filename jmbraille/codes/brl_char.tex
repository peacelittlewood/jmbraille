%%
%% \brl@char@breakable#1#2#3
%%
%% #1:mode
%% #2:penalty : never -> automatically ignorespaces
%% #3:spaces
%%
%% if #2 is lineend, then break here and restart.
%%      and  {#1} = {{label}{leader}{leadersp}{leastleadernum}} or {}
%%
%% if #2 is block, then take it into account in the last process -> see (*)
%%
%% if #2 is never or first, then all the rest breakable are ignored.
%%
%% in math mode, breakable does not give rise to spaces
%% and spaces in text are not meaningful ones (not counted).
%%
%% algorithm : #2, #3
%%
%% if #2=lineend or break
%%   restart
%% else
%%   if #2>\z@ (break_never),
%%   then
%%     ignorespaces and ignore all the consecutive breakable.
%%   else
%%     add spaces
%%
%% (*) after all, if the break is never and there is break_block, it is replaced by break_almostnever.
%% this is to avoid "braille too long."
%% first = never except that it is not replaced by almostnever.
%%
%%       

% #2 lineend < break < bigger < ... < small < almostnever < never < first

%%   \brl@text_dospecial_\brl@text_cmd_dollar (math in text. Only check whether there is a meaningful space.)
%%   \brl@text_alph_subM (math in text. Only check whether there is a meaningful space.)
%%   \brl@math_matrix_prep_rowcol_element (matrix element. Only check whether there is a meaningful space.)

%% and spaces in text are not meaningful ones (not counted).

\def\brl@char_space#1{%
  \brl@char@spacefalse%
  {%
    \let\brl@do\brl@do_space%
    \let\brl@char@cmd_direct\@gobble%
    % \block is #4 (default).
    % \breakable is ignored (default).
    %
    #1%
    %
    \ifbrl@char@space%
      \aftergroup\brl@char@spacetrue%
    \fi%
  }%
}

%%
%% \brl@char_count
%%
%% input: chars
%%
%% output:
%%   \ifbrl@char@space: if brl@toks contains chars with some spaces
%%   \brl@char@charnum: the total number of chars
%%
%% caller:
%%   \brl@print_shipout (nonbraille page number. breakable is meaningless and should be ignored.)
%%   \brl@format_concat_appendlinetoks (breakable is already treated, that is, no breakable)
%%   \rectbox
%%   \ovalbox
%%   \frame
%%
%%   no lineend
%% 
\newcount\brl@char@charnum
\def\brl@char_count#1{%
  {%
    \brl@char@charnum\z@%
    \let\brl@do\brl@do_count%
    \let\brl@char@cmd_direct\@gobble%
    \def\brl@char@block{\brl@char@block_count}%
    \def\brl@char@breakable{\brl@char@breakable_count}%
    %
    #1%
    %
    \xdef\brl@tmp{\the\brl@char@charnum}%
  }%
  \brl@char@charnum\brl@tmp\relax%
}


\def\brl@char@block_count#1#2#3#4#5#6{%
  \advance\brl@char@charnum#3\relax%
}

\def\brl@char@breakable_count#1#2#3{%
  \ifnum#3>\z@%
    \advance\brl@char@charnum#3\relax%
  \fi%
}%

}

%% output : \brl@char@toks, \brl@char@spacetrue or false
\def\brl@char_user#1-{%
  \def\brl@tmp{#1}%
  \ifx\brl@tmp\brl@@end%
  \else%
    \advance\brl@char@charnum\@ne%
    %
    \def\brl@char_user@bca{0}%
    \def\brl@char_user@bcb{0}%
    \count@\z@%
    % 
    \brl@char_user_setbit#1\brl@end%
    %
    \edef\brl@char_user@two{\brl@char_user@bca\brl@char_user@bcb}%
    \ifx\brl@char_user@two\brl@char_user@empty%
      \edef\brl@tmp{{\noexpand\brl@bc{\the\count@}}}%
      % \ifnum\count@=\z@% space
      \ifcase\count@% space
        \brl@char@spacetrue%
      \fi%
    \else%
      \edef\brl@tmp{{\noexpand\brl@ebc{\the\count@}\brl@char_user@two}}%
    \fi%
    \expandafter\brl@char_appendtoks\brl@tmp%
    % 
    \expandafter\brl@char_user%
  \fi%
}

\def\brl@char_user_setbit#1{%
  \def\brl@tmp{#1}%
  \ifx\brl@tmp\brl@@end%
  \else%
    \ifnum#1=7\relax%
      \def\brl@char_user@bca{1}%
    \else%
      \ifnum#1=8\relax%
        \def\brl@char_user@bcb{1}%
      \else%
        \advance\count@\csname brl@char_user@bc#1\endcsname\relax%
      \fi%
    \fi%
    \expandafter\brl@char_user_setbit%
  \fi%
}
