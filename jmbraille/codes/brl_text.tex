\protected\def\brl@text_jp#1{% we assume that #1 is already expanded.
  %%
  %%
  \def\brl@tmp{#1}%\show\brl@tmp%
  \ifx\brl@tmp\empty%
  \else%
    \let\brl@tmpmat\empty%
    {%
      {%
        \let\brl@text_cmd_empty\brl@@empty%
        \def\brl@text_cmd_glue{\ }%
        %
        % use ..._save to shorten virtualchars in detokenized strings.
        %
        \let\brl@do\brl@do_save%
        \let\brl@char@block\brl@char@block_save%
        \let\brl@text_cmd_virtualchar\brl@text_cmd_virtualchar_save%
        \let\brl@char@space@true\brl@char@space@true_save%
        %
        \xdef\brl@tmp{#1}%
      }%
      \let\brl@text@csname\brl@tmp%
      \edef\brl@text@origcs{brl@text@orig_\detokenize\expandafter{\brl@text@csname}}%
      %
      \ifcsname\brl@text@origcs\endcsname%
        \expandafter\let\expandafter\brl@tmp\csname\brl@text@origcs\endcsname%
        \immediate\write\brl@text@file{%
          \noexpand\expandafter%
          \gdef%
          \noexpand\csname%
          brl@text@orig_\noexpand\detokenize%
          {\unexpanded\expandafter{\brl@text@csname}}%
          \noexpand\endcsname%
          {\unexpanded\expandafter{\brl@tmp}}%
        }%
        \brl@tmp%
      \else%
        \brl@mecab{#1}% mecab
        %% 
        % 
        \brl@text@toks{}%
        \brl@text@prevdigitfalse%
        \brl@text@prevalphfalse%
        \def\brl@text@glue{1}% default glue
        \expandafter\brl@text_loop\brl@text@text\brl@end%
        %
        \xdef\brl@tmp{\the\brl@text@toks}% formatted texts
        % 
        \xdef\brl@tmpmat{\the\brl@text@mattoks}%
        \global\brl@text@mattoks{}%
        % 
        \let\brl@do\brl@do_save%
        \let\brl@char@cmd_direct\brl@char@cmd_direct_save%
        \let\brl@char@block\brl@char@block_save%
        \let\brl@char@breakable\brl@char@breakable_save%
        \let\brl@char@space@true\brl@char@space@true_save%
        %
        \immediate\write\brl@text@file{%
          \noexpand\expandafter%
          \gdef%
          \noexpand\csname%
          brl@text@orig_\noexpand\detokenize%
          {\unexpanded\expandafter{\brl@text@csname}}%
          \noexpand\endcsname%
          {%
            \xdef\noexpand\brl@tmp%
            {%
              \noexpand\brl@text_load%
              \expandafter\brl@text_save\brl@tmp\empty%
              Z%
            }%
            \ifx\brl@tmpmat\empty%
            \else%
              \xdef\noexpand\brl@tmpmat%
              {%
                \expandafter\brl@text_save_matrix\brl@tmpmat\brl@end%
              }%
            \fi%
            % appendmath
            \ifbrl@text_appendmath@topeq%
              \noexpand\brl@text_appendmath@topeqtrue%
            \fi%
            % glue
            \def\noexpand\brl@text@glue{\brl@text@glue}%
          }%
        }%
      \fi%
      \expandafter\brl@document_appendmattoks\expandafter{\brl@tmpmat}%
      \ifbrl@text_appendmath@topeq%
        \global\brl@document@indentfalse% to begin with an equation, force noindent.
      \fi%
      \xdef\brl@tmpglue{\brl@text@glue}%
    }%
    \let\brl@text@glue\brl@tmpglue% used only in Braille Remark.
  \fi%
  \brl@text@toks\expandafter{\brl@tmp}%
}

%% \brl@text_direct does neither expand nor mecab anything.
%%
%% there appears no mattoks
%%               no \brl@text_appendmath@topeqtrue
\protected\def\brl@text_direct#1{%
  %%
  %%
  {%
    \edef\brl@text@origcs{brl@text@mcborig_\detokenize{#1}}%
    \ifcsname\brl@text@origcs\endcsname%
      \expandafter\let\expandafter\brl@tmp\csname\brl@text@origcs\endcsname%
      \immediate\write\brl@text@file{%
        \noexpand\expandafter%
        \gdef%
        \noexpand\csname%
        brl@text@mcborig_\noexpand\detokenize%
        {\unexpanded{#1}}%
        \noexpand\endcsname%
        {\unexpanded\expandafter{\brl@tmp}}%
      }%
      \brl@tmp%
    \else%
      % 
      \brl@text@toks{}%
      \brl@text@prevdigitfalse%
      \brl@text@prevalphfalse%
      \def\brl@text@glue{1}% default glue
      \brl@text_loop#1\brl@end%
      %
      %
      % \brl@format_unbreak{\brl@text@toks}%
      % \xdef\brl@tmp{\the\brl@format@toks}%
      \xdef\brl@tmp{\the\brl@text@toks}%
      %
      \let\brl@do\brl@do_save%
      \let\brl@char@cmd_direct\brl@char@cmd_direct_save%
      \let\brl@char@block\brl@char@block_save%
      \let\brl@char@breakable\brl@char@breakable_save%
      \let\brl@char@space@true\brl@char@space@true_save%
      %
      \immediate\write\brl@text@file{%
        \noexpand\expandafter%
        \gdef%
        \noexpand\csname%
        brl@text@mcborig_\noexpand\detokenize%
        {\unexpanded{#1}}%
        \noexpand\endcsname%
        {%
          \xdef\noexpand\brl@tmp%
          {%
            \noexpand\brl@text_load%
            \expandafter\brl@text_save\brl@tmp\empty%
            Z%
          }%
        }%
      }%
    \fi%
  }%
  \brl@text@toks\expandafter{\brl@tmp}%
}
 

\def\brl@text_save#1{%
  \expandafter\empty#1% expand forcibly
}

\def\brl@do_save#1{%
  {#1}%
  \brl@text_save%
}

\protected\def\brl@char@space@true_save{%
  R%
  \brl@text_save%
}

\def\brl@text_cmd_virtualchar_save#1{%
  U{\brl@text_save#1\empty}%
}

\protected\def\brl@char@breakable_save#1#2#3{%
  \ifnum\brl@char@break_lineend=#2 %
    \brl@text_save_breakable#1\brl@end{#2}{#3}%
  \else%
    V{#1}{#2}{#3}%
  \fi%
  \brl@text_save%
}

\def\brl@text_save_breakable#1{%
  \ifx#1\brl@end% empty
    V{}%
  \else%
    W{\brl@text_save#1\empty}%
    \expandafter\brl@text_save_breakable_label%
  \fi%
}
\def\brl@text_save_breakable_label#1#2#3\brl@end{%
  {\brl@text_save#1\empty}{#2}{#3}%
}

\protected\def\brl@char@block_save#1#2#3#4#5#6{%
  X{#1}{#2}{#3}{#4}{#5}{\brl@text_save_block#6\brl@end\brl@end\brl@end}%
  \brl@text_save%
}
\def\brl@text_save_block#1#2#3{%
  \ifx\brl@end#1%
  \else%
    {#1}{#2}{\brl@text_save#3\empty}%
    \expandafter\brl@text_save_block%
  \fi%
}

\protected\def\brl@char@cmd_direct_save#1{%
  Y\unexpanded{{#1}}%
  \brl@text_save%
}

\def\brl@text_save_matrix#1{%
  \ifx#1\brl@end%
  \else%
    {%
      \noexpand\brl@text_load%
      \brl@text_save#1\empty%
      Z%
    }%
    \expandafter\brl@text_save_matrix%
  \fi%
}

\def\brl@text_loop#1{%
  %% determine: 
  %% \brl@text_next
  %% \brl@text@prevdigittrue or false
  %% \brl@text@prevalphtrue or false
  %% \brl@text@glue
  \edef\brl@text@tmp{brl@text_dospecial_\string#1}%
  \ifcsname\brl@text@tmp\endcsname% dospecial
    \ifcsname brl@text_setglue_\string#1\endcsname% set glue = ignore default glue
    \else%
      \def\brl@text@glue{1}% default glue
    \fi%
    \csname\brl@text@tmp\expandafter\endcsname%
  \else% not dospecial
    \ifbrl@text@prevdigit%
      \brl@text@prevdigitfalse%
      \ifcsname brl@text_prevdigit@\string#1\endcsname%
        \brl@text_appendtoks{\brl@code_text_tsunagi}%
      \fi%
    \else%
      \ifbrl@text@prevalph%
        \brl@text@prevalphfalse%
        \ifcsname brl@text_prevalph@\string#1\endcsname%
          \brl@text_appendtoks{\brl@code_text_tsunagi}%
        \fi%
      \fi%
    \fi%
    \expandafter\brl@text_appendtoks\expandafter{\csname brl@code_text_\string#1\endcsname}%
    \def\brl@text@glue{1}% default glue
    \expandafter\brl@text_loop%
  \fi% dospecial or not
}

      \brl@text@prevalphfalse%
      \brl@text_loop%
