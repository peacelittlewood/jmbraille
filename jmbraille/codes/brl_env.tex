%%
%% origsetting : the setting in this article
%% prevsetting : the setting in the previous article (the article may have been changed).
%% recscale, recorient, recwidth : the setting calculated in the previous article.
%%

\def\brl@env_check_before{%
  \let\brl@figure@origscale\brl@figure@scale%
  \edef\brl@env_check@origsetting{% the setting defined in the current article
    {\brl@figure@scale}%
    {\ifbrl@figure@landscape l\else p\fi}%
    {\ifbrl@figure@spread d\else s\fi}%
  }%
  \let\brl@env_check@prevsetting\empty%
  \ifbrl@figure@autosetting%
    \expandafter\brl@env_check_before_start%
  \fi%
}%

\def\brl@env_check_before_start{%
  \ifcsname brl@var@fig\the\brl@figure@figurenum\endcsname% if exists, the previous setting is not appropreate.
    \edef\brl@env_check@setting{\csname brl@var@fig\the\brl@figure@figurenum\endcsname}%
    %
    \expandafter\brl@env_check_before_\brl@env_check@setting% renew prevsetting
    \ifx\brl@env_check@prevsetting\brl@env_check@origsetting% compare the settings between the present article and the last article -> article is not renewed.
      \PackageWarning{jmbraille}{The figure \the\brl@figure@figurenum\brl@@space may still be out of range. The recommended setting is applied}%
      %
      \BrailleSetFigureScale{\brl@env_check@recscale}%
      %
      \if l\brl@env_check@recorient%
        \BrailleFigureLandscapeOn%
      \else%
        \BrailleFigureLandscapeOff%
      \fi%
      %
      \if d\brl@env_check@recwidth%
        \BrailleFigureSpreadOn%
      \else%
        \BrailleFigureSpreadOff%
      \fi%
    \fi%
  \fi%
}


\def\brl@env_check_before_#1#2#3#4#5#6{%
  \def\brl@env_check@prevsetting{{#1}{#2}{#3}}%
  \def\brl@env_check@prevrecsetting{{#4}{#5}{#6}}%
  \def\brl@env_check@recscale{#4}%
  \def\brl@env_check@recorient{#5}%
  \def\brl@env_check@recwidth{#6}%
}


% \dim
% \z@ : width of picture
% \@ne : height of picture
% \tw@ : width of draw area
% \thr@@ : height of draw area
% 

\def\brl@env_check_after_init{%
  \edef\brl@env_check@ht{\the\dimexpr\brl@figure@scale\dimexpr\ht\brl@env_draw@pictureboxb+\dp\brl@env_draw@pictureboxb\relax+\dimexpr\brl@braille@charheight\relax\relax}%
  \edef\brl@env_check@wd{\the\dimexpr\brl@figure@scale\wd\brl@env_draw@pictureboxb\relax}%
  % 
  \ifbrl@figure@landscape% rotate
    \dimen\z@\brl@env_check@ht\relax% width
    \dimen\@ne\brl@env_check@wd\relax% height
  \else%
    \dimen\z@\brl@env_check@wd\relax% width
    \dimen\@ne\brl@env_check@ht\relax% height
  \fi%
  \ifbrl@figure@spread%
    \dimen\tw@\brl@figure@doublepage%
  \else%
    \dimen\tw@\brl@figure@singlepage%
  \fi%
  \dimen\thr@@\dimexpr\brl@figure@textheight-\dimexpr\brl@braille@charheight\relax\relax% header
}

\def\brl@env_check_after{%
  %
  \brl@env_check_after_init%
  % 
  %
  % \showthe\dimen\z@
  % \showthe\dimen\@ne
  % \showthe\dimen\tw@
  % \showthe\dimen\thr@@
  % 
  % 
  % \brl@env_check_rec% out of range
  % \dim
  % \z@ : width of picture
  % \@ne : height of picture
  % \tw@ : width of draw area
  % \thr@@ : height of draw area
  %
  %
  % if the original setting is OK -> finish.
  %
  % otherwise, calculate a recommendable setting :
  % rotation if nessesary.
  %   single : 
  %   if ratio of shrink is not too small -> finish.
  %     double :
  %     shrink -> finish.
  %
  %
  \brl@env_check@recommendedfalse%
  % 
  \brl@env_check_origsetting% check the original setting
  %
  \ifbrl@env_check@recommended% the original setting falied or the previous recommended setting was applied.
  \else% need to modify
    %
    %   
    \brl@env_check_rec_scale%
    %
    % calculate a recommendable setting based on the current setting.
    %
    \brl@env_check_rec_singlescale% scale in single page.
    % 
    \ifbrl@env_check@recommended%
    \else%
      \brl@env_check_rec_doublescale% scale in double page.
    \fi%
    % 
    % 
    \edef\brl@tmp{%
      {\brl@env_check@recscale}%
      {\brl@env_check@recorient}%
      {\brl@env_check@recwidth}%
    }%
    % 
    % 
    \ifx\brl@tmp\brl@env_check@prevrecsetting% recommended setting is OK.
      \PackageNote{jmbraille}{The recommended setting seems good}%
    \else%
      \ifbrl@figure@autosetting%
        \PackageWarning{jmbraille}{The figure \the\brl@figure@figurenum\brl@@space is out of range. The recommended setting will be applied in the next time if it is not changed manually}%
      \else%
        \PackageWarning{jmbraille}{The figure \the\brl@figure@figurenum\brl@@space is out of range. Change the setting manually}%
      \fi%
    \fi%
    \PackageNote{jmbraille}{The recommended setting is^^J%
      \string\BrailleSetFigureScale{\brl@env_check@recscale}^^J%
      \string\BrailleFigureLandscape\if l\brl@env_check@recorient On\else Off\fi^^J%
      \string\BrailleFigureSpread\if d\brl@env_check@recwidth On\else Off\fi^^J}%
    % 
    %
    \if@filesw%
      \immediate\write\@auxout{%
        \global\noexpand\@namedef{brl@var@fig\the\brl@figure@figurenum}{%
          \brl@env_check@origsetting%
          \brl@tmp%
        }%
      }%
    \fi%
  \fi%
}
%


\def\brl@env_check_origsetting{%
  \ifx\brl@env_check@prevsetting\empty% the original setting was applied.
    \ifnum\dimen\z@<\dimen\tw@%
      \ifnum\dimen\@ne<\dimen\thr@@%
        \brl@env_check@recommendedtrue% the original setting is OK.
      \fi%
    \fi%
  \fi%
}

\def\brl@env_check_rec_scale{%
  % rotate if necessary
  %
  \edef\brl@env_check@hta{\the\dimexpr\brl@figure@scale\dimexpr\ht\brl@env_draw@pictureboxb+\dp\brl@env_draw@pictureboxb\relax\relax}%
  \edef\brl@env_check@htb{-\the\dimexpr\brl@braille@charheight\relax}%
  %
  % 
  \ifdim\brl@env_check@wd>\brl@env_check@hta\relax% wd>ht
    \dimen\z@\brl@env_check@hta\relax% width
    \dimen\@ne\brl@env_check@wd\relax% height
    \brl@env_check@rotatedtrue%
    \let\brl@env_check@htx\brl@env_check@htb%
    \let\brl@env_check@hty\empty%
  \else%
    \dimen\z@\brl@env_check@wd\relax% width
    \dimen\@ne\brl@env_check@hta\relax% height
    \let\brl@env_check@hty\brl@env_check@htb%
    \let\brl@env_check@htx\empty%
    \brl@env_check@rotatedfalse%
  \fi%
  %
}

\def\brl@env_check_rec_singlescale{%
  % \dimen\z@ = wd \dimen\@ne = ht, wd<ht
  %
  % \showthe\dimen\z@
  % \showthe\dimen\@ne
  % \showthe\dimexpr\brl@figure@singlepage\relax
  % \showthe\dimen\thr@@
  %
  \edef\brl@env_check@ratiox{\the\dimexpr%
    (\brl@figure@singlepage\brl@env_check@htx)*\p@/\dimen\z@\relax}%
  \edef\brl@env_check@ratioy{\the\dimexpr%
    (\dimen\thr@@\brl@env_check@hty)*\p@/\dimen\@ne\relax}%
  \ifdim\brl@env_check@ratiox>\brl@env_check@ratioy\relax%
    \let\brl@tmp\brl@env_check@ratioy%
  \else%
    \let\brl@tmp\brl@env_check@ratiox%
  \fi%
  %
  %
  %
  \edef\brl@env_check@scale{\the\dimexpr\brl@figure@scale\dimexpr\brl@tmp\relax*\p@/(\dimexpr\brl@figure@origscale\p@\relax)\relax}% compare with the original scale
  %
  \ifdim\brl@env_check@scale>\brl@env_check@tolerance\p@\relax% 80 % -> tolerable
    \brl@env_check@recommendedtrue%
    %
    \ifdim\brl@env_check@scale>\p@\relax% 100 %
      \let\brl@env_check@recscale\brl@figure@origscale% not greater than the original
    \else%
      % 
%      \edef\brl@env_check@recscale{\expandafter\brl@@strip@pt\the\dimexpr0.98\dimexpr\brl@figure@scale\dimexpr\brl@tmp\relax\relax\relax}% to avoid chattering, multiply 0.98
      \edef\brl@env_check@recscale{\expandafter\brl@@strip@pt\the\dimexpr\brl@figure@scale\dimexpr\brl@tmp\relax\relax}%
    \fi%
    \def\brl@env_check@recwidth{s}%
    \edef\brl@env_check@recorient{\ifbrl@env_check@rotated l\else p\fi}%
  \fi%
}

% input : \brl@document@toks and \brl@document@hboxa (tikz or tabular)
% if both are empty, \brl@document@emptyboxtrue
%
% output: \brl@env@toks <- {\brl@env_draw_tabular_print{XXX}}
%
% notice : \egroup corresponding to \bgroup in \brl_document
%
\def\brl@env_draw_tabular_appendtoks{%
  % \brl@document@prevtext must be empty
  \brl@document_exec_text% prevtext must be empty.
  %
  %
  % \print_allsclines_draw yields \print@box
  %
  \brl@print{0}{0}{\brl@text@toks}{\brl@bse@width}%
  %
  % append \print@box to \document@hbox
  %
  % 
  %
  % if \document@hbox is not empty, make \brl@env_draw_tabular@box
  \ifbrl@document@emptybox%
    %
    \let\brl@tmp\empty%
  \else%
    %
    % in picture, there are only hbox. Thus, \brl@document@box must be void.
    %
    \setbox\brl@document@hboxa\hbox{\unhbox\brl@document@hboxa\box\brl@print@boxa}% hbox
    \setbox\brl@document@hboxb\hbox{\unhbox\brl@document@hboxb\box\brl@print@boxb}% hbox
    \setbox\brl@document@boxa\vtop{\unvbox\brl@document@boxa\box\brl@document@hboxa}%
    \setbox\brl@document@boxb\vtop{\unvbox\brl@document@boxb\box\brl@document@hboxb}%
    % 
    \global\advance\brl@env_draw_tabular@boxnum\@ne%
    \edef\brl@tmp{brl@env_draw_tabular@box\the\brl@env_draw_tabular@boxnum}%
    % 
    \ifcsname\brl@tmp a\endcsname%
    \else%
      \expandafter\newbox\csname\brl@tmp a\endcsname%%
      \expandafter\newbox\csname\brl@tmp b\endcsname%%
    \fi%
    % 
    \global\setbox\csname\brl@tmp a\endcsname\box\brl@document@boxa%
    \global\setbox\csname\brl@tmp b\endcsname\box\brl@document@boxb%
    % 
    % 
    \global\brl@document@emptyboxtrue% The boxes are used. Thus reset the switch \brl@document@emptybox.
    %
    \edef\brl@tmp{\brl@env_appendtoks{{\brl@env_draw_tabular_print{\the\brl@env_draw_tabular@boxnum}}}}%
    % \expandafter\brl@tmp%
  \fi%
  \expandafter%
  \egroup% {...} <-
  % all document@box, hbox are reset.
  \brl@tmp% <- \append\tabular_print or empty
}
