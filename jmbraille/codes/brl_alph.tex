\def\brl@alph_appendtoks#1{\brl@alph@toks\expandafter{\the\brl@alph@toks#1}}
\newtoks\brl@alph@toks

\def\brl@alph_appendUtoks#1{\brl@alph@Utoks\expandafter{\the\brl@alph@Utoks#1}}
\newtoks\brl@alph@Utoks

%
% output: \brl@text@toks
%

\protected\def\brl@alph#1{%
  %
  \brl@alph@toks{}%
  \def\brl@alph@currquote{N}% quotation condition
  %
  \let\brl@alph@accent\empty%
  %
  \brl@alph_loop#1\brl@end%
}


\def\brl@alph_loop#1{%
  \csname brl@alph_sub\csname brl@alph@case_\string#1\endcsname\endcsname{#1}%
}

% A : acc
% L : lowercase
% N : number
% O : empty = _
% Q : direct
% S : symbol
% U : uppercase
% W : space
% X : `
% Y : '
% Z : \brl@end

% A
% 
% #1: cmd
% #2: â
% #3: \^
% #4: a
%
% or
%
% #1: cmd
% #2: Æ
% #3: \emptyaccent
% #4: AE
%

\def\brl@alph_subA#1#2#3#4{% accent
  \def\brl@tmp{#3}%
  \ifx\brl@tmp\empty%
  \else%
    \def\brl@alph@accent{\brl@code_alph_englishaccent}% english accent or empty
    %
  \fi%
  \brl@alph_loop#4%
}

\newif\ifbrl@alph@Ufirst

\def\brl@alph_subU#1{% A etc.
  \edef\brl@tmp{{\brl@alph@accent\csname brl@code_alph_#1\endcsname}}%
  %
  \let\brl@alph@accent\empty%
  %
  \brl@alph@Utoks\brl@tmp%
  %
  \brl@alph@Ufirsttrue%
  % 
  \brl@alph_subU_loop%
}

\def\brl@alph_subU_loop#1{% A etc.
  %
  \edef\brl@alph@currcase{\csname brl@alph@case_\string#1\endcsname}%
  \if U\brl@alph@currcase%
    %
    \brl@alph@Ufirstfalse%
    \edef\brl@tmp{{\brl@alph@accent\csname brl@code_alph_#1\endcsname}}%
    \let\brl@alph@accent\empty%
    \expandafter\brl@alph_appendUtoks\brl@tmp%
    \expandafter\brl@alph_subU_loop%
  \else%
    \if A\brl@alph@currcase%
      \def\brl@alph_next{\brl@alph_subU_subA}%
    \else%
      \def\brl@alph_next{\brl@alph_subU_finish#1}%
    \fi%
    \expandafter\brl@alph_next%
  \fi%
}


\def\brl@alph_subU_subA#1#2#3{% accent
  \def\brl@tmp{#2}%
  \ifx\brl@tmp\empty%
  \else%
    \def\brl@alph@accent{\brl@code_alph_englishaccent}% english accent or empty
    %
  \fi%
  \brl@alph_subU_loop#3%
}


\def\brl@alph_subU_finish{%
  %
  % \ifnum\brl@alph@Unum=\@ne%
  \ifbrl@alph@Ufirst%
    \def\brl@alph@Utop{\brl@code_text_capital}%
    \let\brl@alph@Ulast\empty%
  \else%
    \def\brl@alph@Utop{\brl@code_text_doublecapital}%
    \if L\brl@alph@currcase%
      \def\brl@alph@Ulast{\brl@alph@small}% append stopcapital or small
    \else%
      \let\brl@alph@Ulast\empty%
    \fi%
  \fi%
  %
  \edef\brl@tmp{{\brl@alph@Utop\the\brl@alph@Utoks\brl@alph@Ulast}}%
  \expandafter\brl@alph_appendtoks\brl@tmp%
  \brl@alph_loop%
}

\def\brl@alph_subL#1{% a etc.
  \edef\brl@tmp{{\brl@alph@accent\csname brl@code_alph_#1\endcsname}}%
  \let\brl@alph@accent\empty%
  %
  \expandafter\brl@alph_appendtoks\brl@tmp%
  %
  %
  \brl@alph_loop%
}


\def\brl@alph_subS#1{% ; etc.
  \edef\brl@tmp{{\csname brl@code_alph_\string#1\endcsname}}%
  \expandafter\brl@alph_appendtoks\brl@tmp%
  \brl@alph_loop%
}

\def\brl@alph_subN#1{% 1 etc.
  \edef\brl@tmp{{\brl@code_text_numbersign\csname brl@code_alph_#1\endcsname}}%
  \expandafter\brl@alph_appendtoks\brl@tmp%
  %
  \brl@alph_numbermode%
  \brl@alph_subN_loop%
}

  
\def\brl@alph_subN_loop#1{% 1 etc.
  %
  \if N\csname brl@alph@case_\string#1\endcsname%
    %
    \edef\brl@tmp{{\csname brl@code_alph_#1\endcsname}}%
    \expandafter\brl@alph_appendtoks\brl@tmp%
    \expandafter\brl@alph_subN_loop%
  \else%
    \brl@alph_alphmode%
    \ifcsname brl@@letteraj_\string#1\endcsname%
      \brl@alph_appendtoks{\brl@code_text_small}%
    \fi%
    \def\brl@alph_next{\brl@alph_loop#1}%
    \expandafter\brl@alph_next%
  \fi%
}

%% special commands
%%


\def\brl@alph_subQ#1#2{% cmd_direct
  \brl@alph_appendtoks{\brl@char@cmd_direct{#2}}%
  \brl@alph_loop%
}


\def\brl@alph_subO#1{% _
  \brl@alph_loop%
}

\def\brl@alph_subW#1{% glue etc.
  \brl@alph_subW_loop%
}

\def\brl@alph_subW_loop#1{% skip consecutive W and omit the spaces at the end.
  %
  \if W\csname brl@alph@case_\string#1\endcsname%
    \expandafter\brl@alph_subW_loop%
  \else%
    \if Z\csname brl@alph@case_\string#1\endcsname% end
    \else%
      \edef\brl@tmp{{\brl@char@breakable{t}{\brl@char@break_big}{1}}}%
      \expandafter\brl@alph_appendtoks\brl@tmp%
    \fi%
    \def\brl@alph_next{\brl@alph_loop#1}%
    \expandafter\brl@alph_next%
  \fi%
}




\def\brl@alph_subX#1{%
  \def\brl@alph@currquote{S}%
  \brl@alph_appendtoks{\brl@code_alph_singlequoteopen}%
  \brl@alph_loop%
}

\def\brl@alph_subY#1#2{%
  \if N\brl@alph@currquote% not between quotation mark.
    \brl@alph_appendtoks{\brl@code_alph_apostrophe}%
  \else%
    \ifcsname brl@alph@prime_\csname brl@alph@case_\string#2\endcsname\endcsname% check the next letter
      \brl@alph_appendtoks{\brl@code_alph_apostrophe}%
    \else%
      \brl@alph_appendtoks{\brl@code_alph_singlequoteclose}%
      \def\brl@alph@currquote{N}%
    \fi%
  \fi%
  \brl@alph_loop#2%
}

% ' is apostrophe before the following
% 's 'S '18 etc.
\@namedef{brl@alph@prime_L}{}
\@namedef{brl@alph@prime_U}{}
\@namedef{brl@alph@prime_N}{}



\def\brl@alph_subZ#1{% \brl@end
}
