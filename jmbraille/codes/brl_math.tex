%
% input : contents, breaklevel
%
% output : \brl@math@subtoks
%          \brl@math@subid = first = last if elenum = 1 otherwise null
%          \brl@math@subfirstid
%          \brl@math@sublastid

\def\brl@math_parse#1{%
  {%
    % \def\brl@tmp{#1}\show\brl@tmp
    %
    % the following level may be broken in math_exec
    % due to the input if the pair () etc. does not match.
    %
    \edef\brl@math@origbracketlevel{\the\brl@math@bracketlevel}%
    \edef\brl@math@origbreaklevel{\the\brl@math@breaklevel}%
    % 
    % preproc init
    %
    \brl@math@posnum\z@%
    \brl@math@bracetoks{}%
    \brl@math@directtoks{}%
    %
    % preproc (pass 1) construct queues
    %                  table of underbrace, overbrace
    %                  env
    % sslevel (breaklevel) may be changed but restored in ^ and _ (except underbrace and overbrace)
    %
    \brl@math_preproc#1\brl@end%
    \edef\brl@math@posmax{\the\brl@math@posnum}%
    %
    % parse init
    %
    \brl@math@posnum\z@%
    \brl@math@elenum\z@% to count the elenum of #1
    % 
    % parse (pass 2) construct brailles
    %
    % bracketlevel, breaklevel may be changed to count the number of elements:
    % text@parenlevel may be changed in () used in \text
    % bracketlevel may be changed in\{\} etc. to restore element numbers (see exec_open)
    %
    % breaklevel is used to determine textmode or mathmode.
    % 
    \brl@math_exec%
    %
    % makebrace (pass 3) construct underbrace, overbrace
    % makedelim (pass 4) construct left, middle, right
    %
    % sslevel and breaklevel are set to be 0 but restored in _{...} or ^{...}
    % 
    % 
    \brl@math_makebrace%
    \brl@math_makedelim%
    %
    % postproc init
    %
    \brl@math@posnum\z@%
    \brl@math@toks{}%
    %
    \brl@math@currblockfalse%
    \brl@math@currbreak\brl@char@break_first% start with break_first
    \brl@math@currspace\z@%
    \def\brl@math@currmode{m}%
    %
    \brl@math@bracketlevel\brl@math@origbracketlevel\relax%
    \brl@math@breaklevel\brl@math@origbreaklevel\relax%
    %
    % postproc (pass 5) insert break
    %                   construct left, right, middle
    %
    %
    \brl@math_log_seg%
    % 
    \brl@math_postproc%
    %
    % firstchar
    \brl@math_get@id{0}{\brl@tmp}%
    \global\let\brl@math@subfirstid\brl@tmp%
    % lastchar
    \brl@math_get@fullid{\the\numexpr\brl@math@posmax+\m@ne\relax}{\brl@tmp}%
    % \ifcsname brl@math_subid@\brl@tmp\endcsname%
    %   \brl@math_get@subid{\the\numexpr\brl@math@posmax+\m@ne\relax}{\brl@tmp}%
    % \fi%
    \global\let\brl@math@sublastid\brl@tmp%
    %
    \ifnum\brl@math@elenum=\@ne%
      \global\let\brl@math@subid\brl@math@sublastid%
    \else%
      \gdef\brl@math@subid{null}%
    \fi%
    %
    \edef\brl@tmp{\the\brl@math@directtoks}% append label
    \ifx\brl@tmp\empty%
    \else%
      \edef\brl@tmp{\expandafter\brl@char@cmd_direct\expandafter{\the\brl@math@directtoks}}%
    \fi%
    %
    \xdef\brl@math@subtoks{\the\brl@math@toks\brl@tmp}%
    \brl@math_log_longseg%
    %
  }%
}

\def\brl@math_env_split#1{%
  {%
    \brl@math@linetoks{}%
    \brl@math_split_parse#1\brl@end%
  }%
  %
  \brl@math_set@iddat{\the\brl@math@posnum}{split}{\brl@tmp}%
  \advance\brl@math@posnum\@ne%
  \brl@math_preproc%
}



\def\brl@math_split_parse#1{%
  %
  \def\brl@math@makebreak_set{\brl@math@makebreak_break}%
  \brl@math_parse{#1}%
  \let\brl@math@lasttoks\brl@math@subtoks%
  \brl@math_split_parse_loop%
}

\def\brl@math_split_parse_loop#1{%
  %
  \def\brl@tmp{#1}% label
  \ifx\brl@tmp\brl@@end%
    %
    \xdef\brl@tmp{\the\brl@math@linetoks\brl@math@lasttoks}%
    %
  \else%
    %
    % \brl@math
    % 
    \brl@math_parse{#1}%
    % 
    % \brl@math@toks
    %
    \ifx\brl@math@subfirstid\brl@math@@arrow%
      \edef\brl@math@lasttoks{\brl@math@lasttoks\brl@code_math_space}% if the char of the newline is arrow, append space at the end of the last line.
      \edef\brl@math@subtoks{\expandafter\brl@@gobblefive\brl@math@subtoks}% delete first space
    \fi%
    % 
    %
    \edef\brl@tmp{{\brl@math@lasttoks\brl@char@breakable{m}{\brl@char@break_break}{0}}}% may
    \expandafter\brl@math_appendlinetoks\brl@tmp%
    %
    \let\brl@math@lasttoks\brl@math@subtoks%
    \expandafter\brl@math_split_parse_loop%
  \fi%
}

\def\brl@math_cases_parse#1{%
  \def\brl@tmp{#1}%
  \ifx\brl@tmp\brl@@end%
    \brl@math_set@htdp{\the\brl@math@posnum}{\brl@tmpheight}{\brl@tmpdepth}%
  \else%

    \brl@math_cases_parse_element#1&&\brl@end%
    \expandafter\brl@math_cases_parse%
  \fi%
}


\def\brl@math_cases_parse_element#1&#2&#3\brl@end{% discard #3
  {%
    %
    \brl@math@sslevel\z@%
    % 
    \brl@math@breaklevel\z@%
    \brl@math_parse{#1}%
    \let\brl@math@lasttoks\brl@math@subtoks%
    \def\brl@tmp{#2}%
    \ifx\brl@tmp\empty%
    \else%
      \brl@math_parse{#2}%
      \edef\brl@math@lasttoks{\brl@math@lasttoks\brl@code_math_space\brl@math@subtoks}%
    \fi%
    %
    %
    \brl@math@toks\expandafter{\brl@math@lasttoks}%
    % 
    \brl@format{2}{-2}{\brl@math@toks}{\the\numexpr\brl@bse@width-4\relax}%
    \xdef\brl@tmp{{\the\brl@format@alltoks}}% formatted toks
    \xdef\brl@tmpdepth{\the\brl@format@totaldepth}%

    \xdef\brl@tmpcharnum{\the\brl@format@allmaxcharnum}%
    \xdef\brl@tmpbvskip{\brl@format@bvskip}%
    \xdef\brl@tmpavskip{\brl@format@avskip}%
  }%
  \ifnum\brl@math@casesnum=\@ne%
    \let\brl@math_cases@height\brl@tmpheight% determined here
    \let\brl@math_cases@depth\brl@tmpdepth%
    \let\brl@math_cases@maxcharnum\brl@tmpcharnum%
    \let\brl@math_cases@bvskip\brl@tmpbvskip% determined here
    \let\brl@math_cases@avskip\brl@tmpavskip%
  \else%
    %
    %
    \let\brl@tmpvskip\empty%

    \ifbrl@char@bvskip% current bvskiptrue
      \def\brl@tmpvskip{+\@ne}%
      \brl@math_appendlinetoks{{1}{0}{}}%
    \else%
      \brl@math_cases@avskip% previous avskiptrue

        \def\brl@tmpvskip{+\@ne}%
        \brl@math_appendlinetoks{{1}{0}{}}%
      \fi%
    \fi%
    %
    \let\brl@math_cases@avskip\brl@tmpavskip%
    %
    %
    \edef\brl@math_cases@depth{\the\numexpr\brl@math_cases@depth+\brl@tmpdepth-\brl@tmpheight\brl@tmpvskip\relax}%
    \ifnum\brl@math_cases@maxcharnum<\brl@tmpcharnum\relax%
      \let\brl@math_cases@maxcharnum\brl@tmpcharnum%
    \fi%
  \fi%
  \expandafter\brl@math_appendlinetoks\brl@tmp%
}

\def\brl@math_align_parse#1#2#3{%
  %
  \def\brl@math@makebreak_set{\brl@math@makebreak_break}%
  \def\brl@math@currmode{m}% default
  \brl@math_parse{#1}%
  \ifx\brl@math@subfirstid\brl@math@@intertext% the char of the newline is intertext
    \let\brl@math@cont_before\brl@math@cont_sentence%
  \else%
    \let\brl@math@cont_before\brl@math@cont_neweq%
  \fi%
  \let\brl@math@lasttoks\brl@math@subtoks%
  \def\brl@math@lastchar{#2}%
  \def\brl@math@lastlabel{#3}%
  \brl@math_align_parse_loop%
}

\def\brl@math_align_parse_loop#1#2#3{%
  %
  \def\brl@tmp{#3}% label
  \ifx\brl@tmp\brl@@end%
    %
    \def\brl@math@makebreak_set{\brl@math@makebreak_lastbreak}%
    \brl@math_makebreak% use \brl@math@cont_before
    %
    \xdef\brl@text@math{{\the\brl@text@toks}}% parse end
    %
  \else%
    %
    % \brl@math
    % 
    \brl@math_parse{#1}%
    % 
    % \brl@math@toks
    %
    \let\brl@math@cont_after\empty% default
    \def\brl@math@nextmode{m}% default
    % 
    \ifcsname brl@math_align_parse@lastchar_\detokenize\expandafter{\brl@math@lastchar}\endcsname%
      \def\brl@math@currmode{t}%
      \let\brl@math@cont_after\brl@math@cont_neweq%
    \fi%
    \ifcsname brl@math_align_parse@subid_\brl@math@subfirstid\endcsname%
      \csname brl@math_align_parse@subid_\brl@math@subfirstid\endcsname%
    \fi%
    %
    \brl@math_makebreak% use \brl@math@cont_before
    %
    \let\brl@math@cont_before\brl@math@cont_after%
    \let\brl@math@currmode\brl@math@nextmode%
    \let\brl@math@lasttoks\brl@math@subtoks%
    \def\brl@math@lastchar{#2}%
    \def\brl@math@lastlabel{#3}%
    \expandafter\brl@math_align_parse_loop%
  \fi%
}
