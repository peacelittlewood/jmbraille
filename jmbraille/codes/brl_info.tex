\def\brl@info_appendtoks#1{\brl@info@toks\expandafter{\the\brl@info@toks#1}}
\newtoks\brl@info@toks

\def\brl@info_appendUtoks#1{\brl@info@Utoks\expandafter{\the\brl@info@Utoks#1}}
\newtoks\brl@info@Utoks


% In the context info, each special char including the space is already changed into cs.
% This routine is also used in texttt and may contain spaces.

% A : acc
% E : env
% I : init
% J : Japanese (= 'not the others')
% L : lowercase
% N : number
% Q : direct
% R : symbol breakable before
% S : symbol
% T : symbol breakable after
% U : uppercase
% W : space
% Z : \brl@end


\protected\def\brl@info#1{%
  %
  \brl@info@toks{}%
  % 
  \def\brl@info@prevcase{I}%
  \let\brl@info@accent\empty%
  %
  \brl@info_loop#1\brl@end%
}

\def\brl@info_get@currcase#1{%
  \edef\brl@tmp{brl@info@case_\string#1}%
  \edef\brl@info@currcase{\ifcsname\brl@tmp\endcsname\csname\brl@tmp\endcsname\else J\fi}%
}

\def\brl@info_loop#1{%
  \brl@info_get@currcase{#1}%
  %
  \ifcsname brl@info@break_\brl@info@prevcase\brl@info@currcase\endcsname% breakable pair
    \edef\brl@tmp{{\brl@char@breakable{i}{\brl@char@break_big}{0}}}%
    \expandafter\brl@info_appendtoks\brl@tmp%
  \fi%
  \csname brl@info_sub\brl@info@currcase\endcsname{#1}%
}

% breakable pair

\let\brl@info@break_LN\empty
\let\brl@info@break_NL\empty
\let\brl@info@break_UN\empty
\let\brl@info@break_NU\empty

% 
% #1: cmd
% #2: â
% #3: \^
% #4: a
%
% or
%
% #1: cmd
% #2: Æ
% #3: \emptyaccent
% #4: AE
%

\def\brl@info_subA#1#2#3#4{% accent
  \def\brl@tmp{#3}%
  \ifx\brl@tmp\empty%
  \else%
    \def\brl@info@accent{\brl@code_info_englishaccent}% english accent or empty
    %
  \fi%
  \let\brl@info@prevcase\brl@info@currcase%
  \brl@info_loop#4%
}


\def\brl@info_subJ#1{% A etc.
  \brl@info@Utoks{#1}
  %
  \brl@info_jpmode%
  \brl@info_subJ_loop%
}

\def\brl@info_subJ_loop#1{% A etc.
  %
  \brl@info_get@currcase{#1}%
  %
  \if J\brl@info@currcase%
    %
    \brl@info_appendUtoks{#1}
    \expandafter\brl@info_subJ_loop%
  \else%
    {% already \brl@documented
      \expandafter\brl@text\expandafter{\the\brl@info@Utoks}%
      \xdef\brl@tmp{{\brl@code_text_textquoteopen\the\brl@text@toks\brl@code_text_textquoteclose}}%
    }%
    %
    \expandafter\brl@info_appendtoks\brl@tmp%
    %
    \brl@info_textmode%
    \def\brl@info_next{\brl@info_loop#1}%
    \expandafter\brl@info_next%
  \fi%
}


\newif\ifbrl@info@Ufirst

\def\brl@info_subU#1{% A etc.
  \edef\brl@tmp{{\brl@info@accent\csname brl@code_info_#1\endcsname}}%
  %
  \let\brl@info@accent\empty%
  \brl@info@Utoks\brl@tmp%
  \brl@info@Ufirsttrue%
  % 
  \brl@info_subU_loop%
}

\def\brl@info_subU_loop#1{% A etc.
  %
  \brl@info_get@currcase{#1}%
  %
  \if U\brl@info@currcase%
    %
    %
    \brl@info@Ufirstfalse%
    \edef\brl@tmp{{\brl@info@accent\csname brl@code_info_#1\endcsname}}%
    \let\brl@info@accent\empty%
    \expandafter\brl@info_appendUtoks\brl@tmp%
    \expandafter\brl@info_subU_loop%
  \else%
    \if A\brl@info@currcase%
      \def\brl@info_next{\brl@info_subU_subA}%
    \else%
      \def\brl@info_next{\brl@info_subU_finish#1}%
    \fi%
    \expandafter\brl@info_next%
  \fi%
}


\def\brl@info_subU_subA#1#2#3{% accent
  \def\brl@tmp{#2}%
  \ifx\brl@tmp\empty%
  \else%
    \def\brl@info@accent{\brl@code_info_englishaccent}% english accent or empty
    %
  \fi%
  \brl@info_subU_loop#3%
}


\def\brl@info_subU_finish{%
  %
  %
  \ifbrl@info@Ufirst%
    \def\brl@info@Utop{\brl@code_info_capital}%
    \let\brl@info@Ulast\empty%
  \else%
    \def\brl@info@Utop{\brl@code_info_doublecapital}%
    \if L\brl@info@currcase%
      \def\brl@info@Ulast{\brl@code_info_small}% append stopcapital or small
    \else%
      \let\brl@info@Ulast\empty%
    \fi%
  \fi%
  %
  \edef\brl@tmp{{\brl@info@Utop\the\brl@info@Utoks\brl@info@Ulast}}%
  \expandafter\brl@info_appendtoks\brl@tmp%
  \let\brl@info@prevcase\brl@info@currcase%
  \brl@info_loop%
}

\def\brl@info_subL#1{% a etc.
  \edef\brl@tmp{{\brl@info@accent\csname brl@code_info_#1\endcsname}}%
  \let\brl@info@accent\empty%
  %
  \expandafter\brl@info_appendtoks\brl@tmp%
  %
  %
  \let\brl@info@prevcase\brl@info@currcase%
  \brl@info_loop%
}

\def\brl@info_subN#1{% 1 etc.
  \edef\brl@tmp{{\brl@code_info_numbersign\csname brl@code_info_#1\endcsname}}%
  \expandafter\brl@info_appendtoks\brl@tmp%
  %
  \brl@info_numbermode%
  \brl@info_subN_loop%
}

  
\def\brl@info_subN_loop#1{% 1 etc.
  %
  \brl@info_get@currcase{#1}%
  \if N\brl@info@currcase%
    \edef\brl@tmp{{\csname brl@code_info_#1\endcsname}}%
    \expandafter\brl@info_appendtoks\brl@tmp%
    \expandafter\brl@info_subN_loop%
  \else%
    \brl@info_textmode%
    \ifcsname brl@@letteraj_\string#1\endcsname%
      \brl@info_appendtoks{\brl@code_info_small}%
    \fi%
    \def\brl@info_next{\brl@info_loop#1}%
    \let\brl@info@prevcase\brl@info@currcase%
    \expandafter\brl@info_next%
  \fi%
}

\def\brl@info_subQ#1#2{% cmd_direct
  \brl@info_appendtoks{\brl@char@cmd_direct{#2}}%
  \brl@info_loop%
}


\def\brl@info_subW#1{% glue etc.
  \brl@info_subW_loop%
}

\def\brl@info_subW_loop#1{% skip consecutive W and omit the spaces at the end.
  %
  \brl@info_get@currcase{#1}%
  % 
  \if W\brl@info@currcase%
    \expandafter\brl@info_subW_loop%
  \else%
    \if Z\brl@info@currcase%
    \else%
      \if I\brl@info@prevcase%
      \else%
        \edef\brl@tmp{{\brl@char@breakable{i}{\brl@char@break_big}{1}}}%
        \expandafter\brl@info_appendtoks\brl@tmp%
      \fi%
    \fi%
    \let\brl@info@prevcase\brl@info@currcase%
    \def\brl@info_next{\brl@info_loop#1}%
    \expandafter\brl@info_next%
  \fi%
}
