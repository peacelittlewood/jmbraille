\def\brl@format#1#2#3#4{%
  \edef\brl@format@leftmargin{#1}%
  \edef\brl@format@indent{#2}%
  \edef\brl@format@width{#4}%
  \edef\brl@format@widthspread{\the\numexpr\tw@*\brl@format@width\relax}%
  \brl@format@alltoks{}% output
  \brl@format@linetoks{}%
  %
  \let\brl@format@bvskip\empty% output
  \let\brl@format@avskip\empty% output
  %
  \brl@format@totalheight\brl@format@firstline\relax% dummy. This means the first line.
  \brl@format@firstvskiptrue%
  %
  \brl@format@allmaxcharnum\z@%
  %
  % \showthe#3
  %
  % \expandafter\brl@format_loop\the#3\brl@char@breakable\brl@end\brl@end\brl@end%
  \expandafter\brl@format_loop\the#3\brl@char@breakable{}Z\brl@end%
  %
  % \showthe\brl@format@alltoks
}



\def\brl@format_loop#1\brl@char@breakable{%
  %
  %
  \def\brl@tmp{#1}%
  % \show\brl@tmp%
  %
  \ifx\brl@tmp\empty% 
    %
    % this happens only when
    % \brl@char@breakable{...}\brl@char@breakable
    %
    % such as \end{proof}, \[...\]
    %
    % in particular, 
    % \[...\] \qquad
    % \[...\]
    % gives rise to ... \breakable\breakable ...
    % the first breakable is the last breakable of eq
    % the second breakable is that in text_dospecial_\cmd_env
    % (see text_dospecial_\cmd_env)
    % -> simply ignore the second \breakable
    %
    %
    % gobble \brl@end\brl@end\brl@end
    %
    \expandafter\brl@format_loop_ignorebreakable%
  \else%
    %
    %
    \brl@char_count{#1}%
%    \brl@format_appenddattoks{{#1}}%
    \brl@format_appenddattoks{#1}%
    %
    \expandafter\brl@format_setpenalty%
  \fi%
}


\def\brl@format_loop_ignorebreakable#1#2#3{%
  \def\brl@tmp{#3}%
  \ifx\brl@tmp\brl@@end% multiple \breakable.
    % exit
  \else%
    \expandafter\brl@format_loop% behave as if there were no second breakable
  \fi%
}

%% format_chunk@prevXXXX : data for the previous line
%% format_chunk@currXXXX : data for the current line
%% format_chunk@nextXXXX : data for the next line

\def\brl@format_appendprevtoks#1{\brl@format@prevtoks\expandafter{\the\brl@format@prevtoks#1}}
\newtoks\brl@format@prevtoks


\newif\ifbrl@format@allowspread


\def\brl@format_chunk_pulltoksws{%
  \expandafter\brl@format_chunk_pulltoksws_\the\brl@format@dattoks\brl@end%
}

\def\brl@format_chunk_pulltoksws_#1#2#3\brl@end{%
  \brl@format@dattoks{#3}%
  \brl@format_appendprevtoks{#1#2}%
}

\def\brl@format_chunk_pulltokswos{%
  \expandafter\brl@format_chunk_pulltokswos_\the\brl@format@dattoks\brl@end%
}

\def\brl@format_chunk_pulltokswos_#1#2#3\brl@end{%
  \brl@format@dattoks{#3}%
  \brl@format_appendprevtoks{#1}%
  \def\brl@format_chunk@space{#2}%
}


%% determine the width. 
%%


\def\brl@format_chunk_widthspread{% first time
  % 
  \edef\brl@format_chunk@bsecharnum{\the\numexpr\brl@format_chunk@bsecharnum_base-\brl@format_chunk@lineend\relax}%
  %
  \ifnum\brl@format_chunk@wordwidth>\brl@format_chunk@bsecharnum\relax% requires double spread
    \let\brl@format_chunk@bsecharnum_base\brl@format@widthspread%
    \brl@format_chunk_width_%
    \def\brl@format_chunk_width{\brl@format_chunk_width_}%
  \fi%
}

\def\brl@format_chunk_width_{%
  \edef\brl@format_chunk@bsecharnum{\the\numexpr\brl@format_chunk@bsecharnum_base-\brl@format_chunk@lineend\relax}%
  \ifnum\brl@format_chunk@wordwidth>\brl@format_chunk@bsecharnum\relax%
    \PackageError{jmbraille}{The braille is too long. Try \string\BrailleBreak}{}%
  \fi%
}

\def\brl@format_chunk{%
  % 
  \ifbrl@format@allowspread%
    \def\brl@format_chunk_width_default{\brl@format_chunk_widthspread}%
  \else%
    \def\brl@format_chunk_width_default{\brl@format_chunk_width_}%
  \fi%
  % 
  %
  \def\brl@format_chunk@prevlinecharnum{\m@ne}%
  \def\brl@format_chunk@currlinestart{0}%
  \edef\brl@format_chunk@currleftmargin{\the\numexpr\brl@format@leftmargin+\brl@format@indent\relax}%
  % 
  %
  \def\brl@format@indent{0}%
  %
  %
  \brl@format_chunk_loop%
}

\def\brl@format_chunk_loop{% start the construction of a line.
  %
  \brl@format@linecharnum\numexpr\brl@format_chunk@currleftmargin+\brl@format_chunk@currlinestart\relax% nonexpr
  %
  \let\brl@format_chunk@linestart\brl@format_chunk@currlinestart%
  \let\brl@format_chunk@leftmargin\brl@format_chunk@currleftmargin%
  %
  % 
  \brl@format@linetoks{}%
  \brl@format@linepenaltypos\z@%
  \brl@format@linepenalty\brl@format@maxpenalty\relax%
  %
  \let\brl@format_chunk@bsecharnum_base\brl@format@width%
  %
  \let\brl@format_chunk_width\brl@format_chunk_width_default%
  %
  \brl@format_chunk_%
}

%% input  : {charnum}{t,m,M,i,Z}{penalty}{spaces}...

\def\brl@format_chunk_#1#2#3#4{%
  %
  \advance\brl@format@linepenaltypos\@ne%
  \brl@format_appendlinetoks{{#1}{#2}{#3}{#4}}%
  %
  \edef\brl@format_chunk@wordwidth{\the\numexpr\brl@format_chunk@leftmargin+\brl@format_chunk@linestart+#1\relax}%
  % 
  % 
  \let\brl@format_chunk@leftmargin\brl@format@leftmargin%
  \edef\brl@format_chunk@linestart{\if#2i\thr@@\else\if#2M\m@ne\else\z@\fi\fi}%
  \edef\brl@format_chunk@lineend{\ifnum\brl@char@break_lineend=#3\z@\else\if#2m\@ne\else\if#2M\@ne\else\z@\fi\fi\fi}%
  %
  %
  \brl@format_chunk_width%
  %
  % 
  \advance\brl@format@linecharnum#1\relax% add number of chars
  % 
  \ifnum\brl@format_chunk@bsecharnum<\brl@format@linecharnum% if width<number of chars
    \def\brl@format_chunk_next{\brl@format_chunk_break}% linebreak
  \else% width>=number of chars
    \ifnum#3>\brl@char@break_break\relax% linebreak
      \ifnum#3>\brl@format@linepenalty%
      \else%
        \edef\brl@format_chunk@maxpos{\the\brl@format@linepenaltypos}%
        \brl@format@linepenalty#3\relax% change min penalty
        %
        %
        %
        \def\brl@format_chunk@currspacenum{#4}%
        \edef\brl@format_chunk@currlinecharnum{\the\brl@format@linecharnum}% save width
        \let\brl@format_chunk@currbsecharnum\brl@format_chunk@bsecharnum_base% 32 or 64
        \let\brl@format_chunk@currlineend\brl@format_chunk@lineend%
        \let\brl@format_chunk@nextlinestart\brl@format_chunk@linestart%
        \brl@format@linetoks{}% clear 
        %
      \fi%
      \advance\brl@format@linecharnum#4\relax% add number of spaces
      \def\brl@format_chunk_next{\brl@format_chunk_}%
    \else% line_break
      \edef\brl@format_chunk@maxpos{\the\brl@format@linepenaltypos}%
      %
      %
      \def\brl@format_chunk@currspacenum{0}%
      \def\brl@format_chunk@currlinecharnum{10000}% save width
      \let\brl@format_chunk@currbsecharnum\brl@format_chunk@bsecharnum_base% 32 or 64
      \let\brl@format_chunk@currlineend\brl@format_chunk@lineend%
      \let\brl@format_chunk@nextlinestart\brl@format_chunk@linestart%
      \brl@format@linetoks{}% clear 
      % 
      \ifnum#3=\brl@char@break_lineend\relax% lineend
        \def\brl@format_chunk@label{#4}% needs for leaders
        \def\brl@format_chunk_next{\brl@format_chunk_end}% lineend
      \else% forced linebreak
        % 
        %
        %
        \def\brl@format_chunk_next{\brl@format_chunk_break}% forced linebreak
      \fi%
    \fi%
  \fi%
  \brl@format_chunk_next%
}


\def\brl@format_chunk_break{%
  \brl@format_chunk_break_sub%
  %
  \expandafter\brl@format_chunk_loop\the\brl@format@linetoks%
}

\def\brl@format_chunk_newline{%
  \brl@format_maketspace{\brl@format_chunk@currleftmargin}%
  \brl@format@prevtoks\expandafter{\brl@format@fillspace}%
  % 
  \ifnum\brl@format_chunk@currlinestart=\thr@@%
    \brl@format_appendprevtoks{\brl@code_info_linebreak}%
  \else%
    \ifnum\brl@format_chunk@currlinestart=\m@ne%
      \brl@format_appendprevtoks{\@gobble}%
    \fi%
  \fi%
  %
  %
  %
  \let\brl@format_chunk@currleftmargin\brl@format@leftmargin% the next is always noindent.
  \let\brl@format_chunk@currlinestart\brl@format_chunk@nextlinestart%
  %
  \brl@format_chunk_newline_append%
}


\def\brl@format_chunk_newline_append{%
%  
  \let\brl@format_chunk@prevspacenum\brl@format_chunk@currspacenum%
  \let\brl@format_chunk@prevlinecharnum\brl@format_chunk@currlinecharnum%
  \let\brl@format_chunk@prevbsecharnum\brl@format_chunk@currbsecharnum% 32 or 64
  \let\brl@format_chunk@prevlineend\brl@format_chunk@currlineend%
%
  \brl@format@linepenaltypos\z@%
  \brl@format_chunk_cons%
}


\def\brl@format_chunk_break_sub{%
  \ifnum\brl@format_chunk@prevlinecharnum=\m@ne%
    \brl@format_chunk_newline%
  \else%
    \count@\numexpr\brl@format_chunk@prevlinecharnum+\brl@format_chunk@prevspacenum+\brl@format_chunk@currlinecharnum-\brl@format_chunk@currleftmargin-\brl@format_chunk@currlinestart\relax%
    %
    \ifnum\count@>\numexpr\brl@format_chunk@prevbsecharnum-\brl@format_chunk@currlineend\relax%
      % line is over, thus the previous line is fixed.
      %
      \ifnum\brl@format_chunk@prevlineend=\@ne%
        \brl@format_appendprevtoks{\brl@code_math_linebreak}% in math mode.
      \fi%
      %
      \brl@format_oneline% <- prevtoks
      %
      %
      \brl@format_calcminmax%
      % 
      \brl@format_chunk_newline%
      % 
    \else%
      %
      \edef\brl@format_chunk@currlinecharnum{\the\count@}%
      \expandafter\brl@format_appendprevtoks\expandafter{\brl@format_chunk@space}%
      \brl@format_chunk_newline_append%
    \fi%
  \fi%
}


\def\brl@format_chunk_end{%
  \brl@format_chunk_break_sub%
  \ifx\brl@format_chunk@label\empty%
    \brl@format_oneline%
  \else%
    \expandafter\brl@format_oneline_withlabel\brl@format_chunk@label%
  \fi%
  %
  %
  \brl@format_calcminmax% exit
  % 
}


\def\brl@format_calcminmax{% 
  \ifnum\brl@format@firstline=\brl@format@totalheight% this means the first line.
    \brl@format@totalheight\brl@format@height%
    \brl@format@totaldepth\brl@format@depth%
    %
    \edef\brl@format@bvskip{\ifbrl@format@bvskip\brl@char@bvskip@true\else\brl@char@bvskip@false\fi}%
    % 
  \else%
    %
    \advance\brl@format@totaldepth\numexpr\brl@format@depth-\brl@format@height\relax%
    %
  \fi%
  % 
  \edef\brl@format@avskip{\ifbrl@format@avskip\brl@char@avskip@true\else\brl@char@avskip@false\fi}%
  % 
  %
  % \showthe\brl@format@alltoks
  % \showthe\brl@format@totalheight
  % \showthe\brl@format@totaldepth
  %% 
  %% 
  %% 
  %% Caution!
  %% output : \brl@format@totalheight <= X < \brl@format@totaldepth
}

\def\brl@format_chunk_cons{%
  \advance\brl@format@linepenaltypos\@ne%
  \ifnum\brl@format_chunk@maxpos=\brl@format@linepenaltypos%
    \brl@format_chunk_pulltokswos%
  \else%
    \brl@format_chunk_pulltoksws%
    \expandafter\brl@format_chunk_cons%
  \fi%
}

\brl@def{brl@format@content_0}{brl@format@currpos_0}\brl@end{%

    \def\brl@format_oneline@charlabel{##1}%
    \def\brl@format_oneline@charleader{##2}%
    \def\brl@format_oneline@leadersp{##3}%
    \def\brl@format_oneline@leastleadernum{##4}%
    %
    %
    \expandafter\brl@format_oneline_\the\brl@format@prevtoks\brl@end%
    % 
    % search a row which can contain the label.
    % 
    \toks@\expandafter{\brl@format_oneline@charlabel}% format
    {%
      \let#1\empty% support only oneline label
      \def#2{0}% support only oneline label
      \brl@format{0}{0}{\toks@}{10000}%
      \expandafter\brl@format_oneline_withlabel_label\the\brl@format@alltoks%
    }%
    \brl@format@scline\z@%
    \def\brl@format_next{\brl@format_oneline_withlabel_}%
    \brl@format_oneline_withlabel_%
    % 
    % 
    \brl@format@scline\brl@format@height%
    % 
    %

\def\brl@format_appendrowdown#1#2#3{%
  %
  \if#1c% center
    \count@\numexpr\brl@format_oneline@currwidth-#2\relax% nonexpr
    \divide\count@\tw@%
  \else% left
    \count@\z@%
  \fi%

  %
  \brl@format@charnum\numexpr\brl@format@maxcharnum-\csname brl@format@currpos_\the\brl@format@scline\endcsname+\count@\relax% number to fill spaces
  % 
  % 
%  \ifnum\brl@format@scline=\z@%
  \ifcase\brl@format@scline%
  \else%
    \ifnum\brl@format@charnum<\z@%
      \brl@format@charnum\z@% too left
    \fi%
    \expandafter\edef\csname brl@format@currpos_\the\brl@format@scline\endcsname{\the\numexpr\csname brl@format@currpos_\the\brl@format@scline\endcsname+\brl@format@charnum+#2\relax}%

  % %
  \advance\brl@format@scline\@ne%
  \ifnum\brl@format_oneline@currdepth>\brl@format@scline%
    \expandafter\brl@format_appendrowdown%
  \fi%
}
