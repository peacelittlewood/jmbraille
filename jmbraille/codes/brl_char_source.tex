\def\brl@char_def_[#1]{%
  \brl@char_def___{#1}%
}
\def\brl@char_def__#1{%
  \brl@char_def___{#1}{#1}%
}

\def\brl@char_def{%
  \let\brl@char@def\brl@char@ndef%
  \let\brl@char@string\empty%
  \@ifnextchar[%
  {\brl@char_def_}%
  {\brl@char_def__}%
}
\def\brl@char_pdef{%
  \let\brl@char@def\brl@char@pdef%
  \def\brl@char@string{\noexpand\string}%
  \@ifnextchar[%
  {\brl@char_def_}%
  {\brl@char_def__}%
}
\def\brl@char_edef{%
  \let\brl@char@def\brl@char@edef%
  \let\brl@char@string\empty%
  \@ifnextchar[%
  {\brl@char_def_}%
  {\brl@char_def__}%
}


%% define ordinary property
\def\brl@char@odef#1{\immediate\write\brl@char@file{\def\expandafter\noexpand\csname#1\brl@char@key\endcsname{\unexpanded\expandafter{\brl@tmp}}}}


%% define char and load
\def\brl@char@ndef#1{%
  \immediate\write\brl@char@file{%
    \protected\def\expandafter\noexpand\csname brl@code_\brl@char@defcontext#1\endcsname{\noexpand\brl@do{\brl@char@key}}%
    ^^J%
    \def\expandafter\noexpand\csname brl@char@load_\brl@char@key\endcsname{%
      \expandafter\noexpand\csname brl@code_\brl@char@defcontext#1\endcsname}%
  }%
}

\def\brl@char@pdef#1{%
  \immediate\write\brl@char@file{%
    \protected\noexpand\expandafter\def\noexpand\csname brl@code_\brl@char@defcontext\unexpanded{\string#1}\noexpand\endcsname{\noexpand\brl@do{\brl@char@key}}%
    ^^J%
    \edef\expandafter\noexpand\csname brl@char@load_\brl@char@key\endcsname{%
      \noexpand\csname brl@code_\brl@char@defcontext\unexpanded{\string#1}\noexpand\endcsname}%
  }%
}

\def\brl@char@edef#1{%
  \immediate\write\brl@char@file{%
    \protected\noexpand\expandafter\def\noexpand\csname brl@code_\brl@char@defcontext\unexpanded{#1}\noexpand\endcsname{\noexpand\brl@do{\brl@char@key}}%
    ^^J%
    \edef\expandafter\noexpand\csname brl@char@load_\brl@char@key\endcsname{%
      \noexpand\csname brl@code_\brl@char@defcontext\unexpanded{#1}\noexpand\endcsname}%
  }%
}



\def\brl@char_def___#1#2#3{%
  %% do
  %
  \edef\brl@char@defcontext{\brl@char@context _}%
  %
  % 
  % 
  %
  \brl@char_getnewkey%
  %
  % deed and load
  \edef\brl@tmp{\noexpand\brl@do{\brl@char@key}}%
  \brl@char@def{#2}%
  %% save
  %
  %% mean
  \def\brl@tmp{#1}%
  \brl@char@odef{brl@char@mean_}%
  %
  %% count
  %% code
  \brl@char@charnum\z@%
  \brl@char@spacefalse%
  \brl@char@toks{}%
  % 
  %
  #3%
  % 
  %
  \edef\brl@tmp{%
    \ifcase\brl@char@charnum%
    \or\advance\brl@char@charnum\@ne%
    \or\advance\brl@char@charnum\tw@%
    \or\advance\brl@char@charnum\thr@@%
    \else\advance\brl@char@charnum\the\brl@char@charnum\relax\fi%
  }%
  %
  \brl@char@odef{brl@char@count_}%
  % 
  \edef\brl@tmp{\ifbrl@char@space\noexpand\brl@char@spacetrue\fi}%
  \brl@char@odef{brl@char@space_}%
  %
  \edef\brl@tmp{\the\brl@char@toks}%
  \brl@char@odef{brl@char@code_}%
  %% 
  %%
  %%
  \@ifnextchar<{\brl@char_def_draw}{\brl@char_def_nodraw}%  
}

% else, \draw is not defined and

%

\def\brl@char_def_draw<#1>{%

}
\def\brl@char_def_nodraw{%
